Option Explicit

Private Const REVIEW_SHEET As String = "同期候補確認履歴"
Private Const CSV_SHEET As String = "目録CSV"
Private Const DEST_SHEET As String = "新ファイル基準表"
Private reviewHeaderCache As Object
Private reviewLastCache As Object
Private reviewManagementIdCounts As Object
Private reviewManagementIdColumn As Long
Private reviewManagementIdIndexReady As Boolean
Private autoReviewBatchDepth As Long
Private autoReviewPending As Boolean
Private autoReviewPendingYear As Long

Public Sub 確認候補を開く()
    Dim rawYear As String, yearValue As Long
    On Error GoTo Failed
    rawYear = Trim$(InputBox("確認する対象年度を入力してください（例：R8）。CSV整形後に実行します。", "同期候補の確認", "R8"))
    If Len(rawYear) = 0 Then Exit Sub
    yearValue = ReviewYear(rawYear)
    If yearValue < 1900 Or yearValue > 2200 Then Err.Raise vbObjectError + 6101, , "対象年度を判定できません。"
    OpenSyncReviewForYear yearValue
    Exit Sub
Failed:
    MsgBox "確認候補を開けません: " & Err.Description, vbExclamation
End Sub

Public Sub SyncReviewBeginBatch()
    If autoReviewBatchDepth = 0 Then
        autoReviewPending = False
        autoReviewPendingYear = 0
    End If
    autoReviewBatchDepth = autoReviewBatchDepth + 1
End Sub

Public Sub SyncReviewAfterStep2Success(ByVal yearValue As Long)
    If yearValue < 1900 Or yearValue > 2200 Then Exit Sub
    If autoReviewBatchDepth > 0 Then
        autoReviewPending = True
        autoReviewPendingYear = yearValue
    Else
        OpenSyncReviewForYear yearValue
    End If
End Sub

Public Sub SyncReviewEndBatch(ByVal allowOpen As Boolean)
    Dim shouldOpen As Boolean, yearValue As Long
    If autoReviewBatchDepth <= 0 Then Exit Sub
    autoReviewBatchDepth = autoReviewBatchDepth - 1
    If autoReviewBatchDepth > 0 Then Exit Sub
    shouldOpen = (allowOpen And autoReviewPending)
    yearValue = autoReviewPendingYear
    autoReviewPending = False
    autoReviewPendingYear = 0
    If shouldOpen Then OpenSyncReviewForYear yearValue
End Sub

Private Sub OpenSyncReviewForYear(ByVal yearValue As Long)
    Dim cases As Collection, copyCases As Collection, conflictCases As Collection
    Dim copyItem As Object, conflictItem As Object, unavailableReason As String
    On Error GoTo Failed
    Set cases = SyncReviewCandidates(yearValue)
    On Error GoTo DUnavailable
    Set copyCases = SyncReviewCopyIdCandidates(yearValue)
    On Error GoTo Failed
    For Each copyItem In copyCases
        cases.Add copyItem
    Next copyItem
    GoTo DisplayCases
DUnavailable:
    If Err.Number <> vbObjectError + 6141 Then GoTo Failed
    unavailableReason = Err.Description
    Resume WithoutD
WithoutD:
    On Error GoTo Failed
DisplayCases:
    Set conflictCases = Module1.BidirectionalSyncConflictReviewCases(yearValue)
    For Each conflictItem In conflictCases
        cases.Add conflictItem
    Next conflictItem
    If cases.Count = 0 Then
        If Len(unavailableReason) > 0 Then MsgBox "D候補を利用できません: " & unavailableReason, vbExclamation
        Exit Sub
    End If
    frmSyncReview.LoadCases cases, yearValue, unavailableReason
    If Not frmSyncReview.Visible Then frmSyncReview.Show vbModeless
    Exit Sub
Failed:
    MsgBox "確認候補を開けません。候補抽出を中止しました: " & Err.Description, vbExclamation
End Sub

Private Function ReviewYearAliasGroup(ByVal aliasName As String) As Long
    Select Case aliasName
        Case "年度（和暦）", "年度(和暦)", "年度": ReviewYearAliasGroup = 1
        Case "作成年度（和暦）", "作成年度(和暦)": ReviewYearAliasGroup = 2
    End Select
End Function

Private Function ReviewStrictAliasYear(ByVal raw As String) As Long
    Dim text As String, c As String, i As Long
    text = UCase$(Trim$(raw))
    text = Replace(text, "令和", "R")
    text = Replace(text, "平成", "H")
    text = Replace(text, "年度", "")
    text = Replace(text, "年", "")
    If Left$(text, 1) = "R" Or Left$(text, 1) = "H" Then text = Mid$(text, 2)
    If Len(text) = 0 Or Len(text) > 10 Then Exit Function
    For i = 1 To Len(text)
        c = Mid$(text, i, 1)
        If AscW(c) < 48 Or AscW(c) > 57 Then Exit Function
    Next i
    ReviewStrictAliasYear = ReviewYear(raw)
End Function

Private Function ReviewCol(ByVal ws As Worksheet, ParamArray aliases() As Variant) As Long
    Dim col As Long, a As Variant, h As String, found As Long, chosen As Long
    Dim cacheKey As String, rowNo As Long, selectedText As String, otherText As String
    Dim yearAliasGroup As Long, selectedYear As Long, otherYear As Long
    Dim yearReferences As Object, yearReferenceCols As Object, referenceCol As Long
    If reviewHeaderCache Is Nothing Then Set reviewHeaderCache = CreateObject("Scripting.Dictionary")
    cacheKey = ws.Name
    ' Year aliases are compared only with aliases from the same family.
    ' This keeps CE's CSV-originated "作成年度" outside review matching.
    If UBound(aliases) >= 1 Then yearAliasGroup = ReviewYearAliasGroup(CStr(aliases(0)))
    For Each a In aliases
        cacheKey = cacheKey & Chr$(30) & CStr(a)
        If yearAliasGroup <> ReviewYearAliasGroup(CStr(a)) Then yearAliasGroup = 0
    Next a
    If yearAliasGroup <> 0 Then
        Set yearReferences = CreateObject("Scripting.Dictionary")
        Set yearReferenceCols = CreateObject("Scripting.Dictionary")
    End If
    If reviewHeaderCache.Exists(cacheKey) Then ReviewCol = CLng(reviewHeaderCache(cacheKey)): Exit Function
    For Each a In aliases
        found = 0
        For col = 1 To ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
            h = Trim$(CStr(ws.Cells(1, col).Value2))
            If StrComp(h, CStr(a), vbTextCompare) = 0 Then
                If found > 0 Then Err.Raise vbObjectError + 6102, , "同一見出しが重複しています: " & h
                found = col
            End If
        Next col
        If found > 0 Then
            If chosen = 0 Then
                chosen = found
            Else
                For rowNo = 2 To ReviewLast(ws)
                    If IsError(ws.Cells(rowNo, chosen).Value2) Or IsError(ws.Cells(rowNo, found).Value2) Then _
                        Err.Raise vbObjectError + 6103, , "別名見出しの値にエラーがあります。"
                    selectedText = Trim$(CStr(ws.Cells(rowNo, chosen).Value2))
                    otherText = Trim$(CStr(ws.Cells(rowNo, found).Value2))
                    If yearAliasGroup <> 0 Then
                        If Not yearReferences.Exists(CStr(rowNo)) And Len(selectedText) > 0 Then
                            yearReferences.Add CStr(rowNo), selectedText
                            yearReferenceCols.Add CStr(rowNo), chosen
                        End If
                        If Len(otherText) > 0 And Not yearReferences.Exists(CStr(rowNo)) Then
                            yearReferences.Add CStr(rowNo), otherText
                            yearReferenceCols.Add CStr(rowNo), found
                        End If
                        If yearReferences.Exists(CStr(rowNo)) Then
                            referenceCol = CLng(yearReferenceCols(CStr(rowNo)))
                            selectedText = CStr(yearReferences(CStr(rowNo)))
                            If Len(otherText) = 0 Then otherText = selectedText
                            selectedYear = ReviewStrictAliasYear(selectedText)
                            otherYear = ReviewStrictAliasYear(otherText)
                            If selectedYear = 0 Or otherYear = 0 Or selectedYear <> otherYear Then _
                                Err.Raise vbObjectError + 6102, , "別名見出しの内容が一致しません: " & ws.Name & " 行" & rowNo & _
                                    " 列" & ws.Cells(1, referenceCol).Address(False, False) & " [" & CStr(ws.Cells(1, referenceCol).Value2) & "] 値=[" & selectedText & "]" & _
                                    " / 列" & ws.Cells(1, found).Address(False, False) & " [" & CStr(ws.Cells(1, found).Value2) & "] 値=[" & otherText & "]"
                        End If
                    ElseIf Len(otherText) > 0 And (Len(selectedText) = 0 Or _
                        StrComp(selectedText, otherText, vbBinaryCompare) <> 0) Then
                        Err.Raise vbObjectError + 6102, , "別名見出しの内容が一致しません: " & ws.Name & " 行" & rowNo
                    End If
                Next rowNo
            End If
        End If
    Next a
    reviewHeaderCache.Add cacheKey, chosen
    ReviewCol = chosen
End Function

Private Function ReviewValue(ByVal ws As Worksheet, ByVal r As Long, ByVal col As Long) As String
    If r < 2 Or col < 1 Then Exit Function
    If IsError(ws.Cells(r, col).Value2) Then Err.Raise vbObjectError + 6103, , "セルにエラー値があります: " & ws.Name & " 行" & r
    ReviewValue = Trim$(CStr(ws.Cells(r, col).Value2))
End Function

Private Function ReviewYear(ByVal raw As String) As Long
    Dim n As Long, text As String, era As String, c As String, i As Long
    text = UCase$(Trim$(raw))
    text = Replace(text, "令和", "R")
    text = Replace(text, "平成", "H")
    text = Replace(text, "年度", "")
    text = Replace(text, "年", "")
    If Len(text) < 1 Then Exit Function
    era = Left$(text, 1)
    If era = "R" Or era = "H" Then text = Mid$(text, 2) Else era = ""
    If Len(text) = 0 Then Exit Function
    If Len(text) > 4 Then Exit Function
    For i = 1 To Len(text)
        c = Mid$(text, i, 1)
        If AscW(c) < 48 Or AscW(c) > 57 Then Exit Function
    Next i
    n = CLng(text)
    Select Case era
        Case "R"
            If n >= 1 And n <= 100 Then ReviewYear = n + 2018
        Case "H"
            If n >= 1 And n <= 31 Then ReviewYear = n + 1988
        Case ""
            If n >= 1900 And n <= 2200 Then ReviewYear = n
    End Select
End Function

Private Function ReviewFiscal(ByVal ws As Worksheet, ByVal r As Long) As Long
    ReviewFiscal = ReviewYear(ReviewValue(ws, r, ReviewCol(ws, "年度（和暦）", "年度(和暦)")))
    If ReviewFiscal = 0 Then ReviewFiscal = ReviewYear(ReviewValue(ws, r, ReviewCol(ws, "年度")))
End Function

Private Function ReviewTextKey(ByVal s As String) As String
    s = Replace(s, vbCr, " "): s = Replace(s, vbLf, " "): s = Replace(s, vbTab, " ")
    s = Replace(s, "　", " "): s = Replace(s, "（", "("): s = Replace(s, "）", ")")
    s = Replace(s, "／", "/"): s = Replace(s, "・", "/"): s = Replace(s, "･", "/")
    On Error Resume Next
    s = StrConv(s, vbNarrow)
    On Error GoTo 0
    ReviewTextKey = LCase$(Replace(Trim$(s), " ", ""))
End Function

Private Function ReviewName(ByVal ws As Worksheet, ByVal r As Long, ByVal isCsv As Boolean, ByVal includeFolder As Boolean) As String
    Dim created As Long, g1 As String, g2 As String, folder As String
    created = ReviewYear(ReviewValue(ws, r, ReviewCol(ws, "作成年度（和暦）", "作成年度(和暦)")))
    If isCsv Then
        g1 = ReviewValue(ws, r, ReviewCol(ws, "分類名２", "分類名2", "第1ガイド", "第１ガイド"))
        g2 = ReviewValue(ws, r, ReviewCol(ws, "分類名３", "分類名3", "第2ガイド", "第２ガイド"))
        folder = ReviewValue(ws, r, ReviewCol(ws, "タイトル", "個別フォルダー", "個別フォルダ"))
    Else
        g1 = ReviewValue(ws, r, ReviewCol(ws, "第1ガイド", "第１ガイド", "分類名２"))
        g2 = ReviewValue(ws, r, ReviewCol(ws, "第2ガイド", "第２ガイド", "分類名３"))
        folder = ReviewValue(ws, r, ReviewCol(ws, "個別フォルダー", "個別フォルダ", "タイトル"))
    End If
    If created = 0 Or Len(g1) = 0 Or Len(g2) = 0 Then Exit Function
    If includeFolder And Len(folder) = 0 Then Exit Function
    ReviewName = CStr(created) & "|" & ReviewTextKey(g1) & "|" & ReviewTextKey(g2)
    If includeFolder Then ReviewName = ReviewName & "|" & ReviewTextKey(folder)
End Function

Private Function ReviewSimilar(ByVal a As String, ByVal b As String) As Boolean
    a = ReviewTextKey(Replace(a, " ", ""))
    b = ReviewTextKey(Replace(b, " ", ""))
    If Len(a) < 5 Or Len(b) < 5 Then Exit Function
    ReviewSimilar = (Left$(a, 5) = Left$(b, 5) Or InStr(1, a, b, vbTextCompare) > 0 Or _
        InStr(1, b, a, vbTextCompare) > 0)
End Function

Private Function ReviewPadding(ByVal s As String) As String
    Dim i As Long, c As String
    If Len(s) = 0 Then Exit Function
    For i = 1 To Len(s)
        c = Mid$(s, i, 1)
        If c < "0" Or c > "9" Then ReviewPadding = s: Exit Function
    Next i
    i = 1
    Do While i < Len(s) And Mid$(s, i, 1) = "0": i = i + 1: Loop
    ReviewPadding = Mid$(s, i)
End Function

Private Function ReviewField(ByVal ws As Worksheet, ByVal r As Long, ByVal fieldName As String) As String
    ReviewField = ReviewValue(ws, r, ReviewCol(ws, fieldName))
End Function

Private Function ReviewLast(ByVal ws As Worksheet) As Long
    Dim cell As Range
    If ws.Name <> REVIEW_SHEET Then
        If Not reviewLastCache Is Nothing Then
            If reviewLastCache.Exists(ws.Name) Then ReviewLast = CLng(reviewLastCache(ws.Name)): Exit Function
        End If
    End If
    Set cell = ws.Cells.Find(What:="*", After:=ws.Cells(1, 1), LookIn:=xlValues, LookAt:=xlPart, _
        SearchOrder:=xlByRows, SearchDirection:=xlPrevious)
    If cell Is Nothing Then ReviewLast = 1 Else ReviewLast = cell.Row
    If ws.Name <> REVIEW_SHEET Then
        If reviewLastCache Is Nothing Then Set reviewLastCache = CreateObject("Scripting.Dictionary")
        reviewLastCache(ws.Name) = ReviewLast
    End If
End Function

Private Function ReviewSignature(ByVal ws As Worksheet, ByVal r As Long) As String
    Dim col As Long, valueText As String, lastCol As Long, signatureLastCol As Long
    If r < 2 Or r > ReviewLast(ws) Then Exit Function
    lastCol = Application.Max(ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column, _
        ws.Cells(r, ws.Columns.Count).End(xlToLeft).Column)
    signatureLastCol = lastCol
    Do While signatureLastCol > 1 And ReviewIsInternalMetadataHeader(ws.Cells(1, signatureLastCol).Value2)
        signatureLastCol = signatureLastCol - 1
    Loop
    ReviewSignature = ws.Name & "|" & CStr(r) & "|" & CStr(signatureLastCol) & "|"
    For col = 1 To lastCol
        If Not ReviewIsInternalMetadataHeader(ws.Cells(1, col).Value2) Then
            If IsError(ws.Cells(r, col).Value2) Then Err.Raise vbObjectError + 6135, , "行にエラー値があります。"
            valueText = TypeName(ws.Cells(r, col).Value2) & ":" & CStr(ws.Cells(r, col).Value2)
            If ws.Cells(r, col).HasFormula Then valueText = valueText & "#" & CStr(ws.Cells(r, col).Formula)
            ReviewSignature = ReviewSignature & CStr(Len(valueText)) & ":" & valueText
        End If
    Next col
    ReviewSignature = ReviewSignature & "|id-format=" & _
        ws.Cells(r, ReviewCol(ws, "ファイルID")).NumberFormat & "|no-format=" & _
        ws.Cells(r, ReviewCol(ws, "ファイル番号")).NumberFormat & "|bar-format=" & _
        ws.Cells(r, ReviewCol(ws, "バーコード番号")).NumberFormat
End Function

Private Function ReviewIsInternalMetadataHeader(ByVal headerValue As Variant) As Boolean
    If IsError(headerValue) Then Exit Function
    If ExcelRowIdIsHeaderName(Trim$(CStr(headerValue))) Then
        ReviewIsInternalMetadataHeader = True
        Exit Function
    End If
    Select Case Trim$(CStr(headerValue))
        Case "最終編集者", "最終編集日時"
            ReviewIsInternalMetadataHeader = True
    End Select
End Function

Private Function ReviewDescription(ByVal ws As Worksheet, ByVal r As Long, ByVal isCsv As Boolean) As String
    Dim nameCol As Long, g1Col As Long, g2Col As Long
    If isCsv Then
        nameCol = ReviewCol(ws, "タイトル", "個別フォルダー")
        g1Col = ReviewCol(ws, "分類名２", "第1ガイド")
        g2Col = ReviewCol(ws, "分類名３", "第2ガイド")
    Else
        nameCol = ReviewCol(ws, "個別フォルダー", "タイトル")
        g1Col = ReviewCol(ws, "第1ガイド", "分類名２")
        g2Col = ReviewCol(ws, "第2ガイド", "分類名３")
    End If
    ReviewDescription = ws.Name & " 行" & r & " / ID=" & ReviewField(ws, r, "ファイルID") & _
        " / 番号=" & ReviewField(ws, r, "ファイル番号") & " / バーコード=" & ReviewField(ws, r, "バーコード番号") & _
        " / 年度=" & CStr(ReviewFiscal(ws, r)) & " / 作成年度=" & ReviewValue(ws, r, ReviewCol(ws, "作成年度（和暦）", "作成年度(和暦)")) & _
        " / 第1=" & ReviewValue(ws, r, g1Col) & " / 第2=" & ReviewValue(ws, r, g2Col) & _
        " / フォルダー=" & ReviewValue(ws, r, nameCol)
    If Not isCsv Then ReviewDescription = ReviewDescription & " / 同期状態=" & ReviewField(ws, r, "同期状態")
End Function

Private Function NewCase(ByVal category As String, ByVal yearValue As Long, ByVal src As Worksheet, _
    ByVal srcRow As Long, ByVal dst As Worksheet, ByVal rows As Collection, ByVal reason As String, _
    Optional ByVal sourceStableKey As String = "") As Object
    Dim item As Object, v As Variant, rowText As String, rowSignatures As String, details As String
    Set item = CreateObject("Scripting.Dictionary")
    For Each v In rows
        rowText = rowText & IIf(Len(rowText) > 0, ";", "") & CStr(v)
        rowSignatures = rowSignatures & ReviewSignature(dst, CLng(v))
        details = details & vbCrLf & ReviewDescription(dst, CLng(v), False)
    Next v
    item.Add "category", category
    item.Add "year", yearValue
    item.Add "sourceRow", srcRow
    item.Add "rows", rowText
    item.Add "reason", reason
    item.Add "sourceSignature", IIf(srcRow > 0, ReviewSignature(src, srcRow), "")
    item.Add "destinationSignature", rowSignatures
    item.Add "destinationIdSnapshot", ReviewManagementIdSnapshot(dst, rows)
    item.Add "sourceStableKey", sourceStableKey
    item.Add "details", IIf(srcRow > 0, ReviewDescription(src, srcRow, True), "CSV対象なし") & vbCrLf & reason & vbCrLf & details
    item.Add "key", category & "|" & CStr(yearValue) & "|" & CStr(srcRow) & "|" & rowText
    Set NewCase = item
End Function

Public Function SyncReviewSelectedDeletionItem(ByVal item As Object, ByVal selectedRow As Long) As Object
    If item Is Nothing Then Exit Function
    If CStr(item("category")) <> "C" Then Exit Function
    If CLng(item("sourceRow")) <> 0 Then Exit Function
    If selectedRow < 2 Then Exit Function
    ' C candidates are deliberately one Excel row per case. Do not turn a
    ' multi-row or stale item into a partial deletion request.
    If CStr(item("rows")) <> CStr(selectedRow) Then Exit Function
    Set SyncReviewSelectedDeletionItem = item
End Function

Private Sub ReviewMapAdd(ByVal map As Object, ByVal key As String, ByVal rowNo As Long)
    Dim rows As Collection
    If Len(key) = 0 Then Exit Sub
    If Not map.Exists(key) Then
        Set rows = New Collection
        map.Add key, rows
    End If
    Set rows = map(key)
    rows.Add rowNo
End Sub

Private Function ReviewRows(ByVal map As Object, ByVal key As String) As Collection
    Dim rows As New Collection
    If Len(key) > 0 Then
        If map.Exists(key) Then Set rows = map(key)
    End If
    Set ReviewRows = rows
End Function

Private Function ReviewYearIndexKey(ByVal yearValue As Long, ByVal valueText As String) As String
    ReviewYearIndexKey = CStr(yearValue) & Chr$(30) & valueText
End Function

Private Function ReviewNormalizeManagementId(ByVal rawValue As Variant, ByRef normalizedId As String) As Boolean
    Dim textValue As String, position As Long, oneChar As String
    normalizedId = ""
    If IsError(rawValue) Then Exit Function
    If IsNull(rawValue) Then Exit Function
    textValue = Trim$(CStr(rawValue))
    If Len(textValue) = 38 And Left$(textValue, 1) = "{" And Right$(textValue, 1) = "}" Then _
        textValue = Mid$(textValue, 2, 36)
    If Len(textValue) <> 36 Then Exit Function
    For position = 1 To 36
        oneChar = Mid$(textValue, position, 1)
        If position = 9 Or position = 14 Or position = 19 Or position = 24 Then
            If oneChar <> "-" Then Exit Function
        ElseIf InStr(1, "0123456789abcdefABCDEF", oneChar, vbBinaryCompare) = 0 Then
            Exit Function
        End If
    Next position
    normalizedId = UCase$(textValue)
    ReviewNormalizeManagementId = True
End Function

Private Sub ReviewBuildManagementIdIndex(ByVal ws As Worksheet)
    Dim rowNo As Long, lastRow As Long, normalizedId As String
    On Error GoTo Failed
    Set reviewManagementIdCounts = CreateObject("Scripting.Dictionary")
    reviewManagementIdColumn = 0
    reviewManagementIdIndexReady = False
    reviewManagementIdColumn = ReviewCol(ws, "Excel管理行ID")
    If reviewManagementIdColumn = 0 Then Exit Sub
    lastRow = ReviewLast(ws)
    For rowNo = 2 To lastRow
        If ReviewNormalizeManagementId(ws.Cells(rowNo, reviewManagementIdColumn).Value2, normalizedId) Then
            If reviewManagementIdCounts.Exists(normalizedId) Then
                reviewManagementIdCounts(normalizedId) = CLng(reviewManagementIdCounts(normalizedId)) + 1
            Else
                reviewManagementIdCounts.Add normalizedId, 1
            End If
        End If
    Next rowNo
    reviewManagementIdIndexReady = True
    Exit Sub
Failed:
    Set reviewManagementIdCounts = Nothing
    reviewManagementIdColumn = 0
    reviewManagementIdIndexReady = False
    Err.Clear
End Sub

Private Function ReviewManagementRowId(ByVal ws As Worksheet, ByVal rowNo As Long) As String
    Dim normalizedId As String
    If Not reviewManagementIdIndexReady Then Exit Function
    If reviewManagementIdCounts Is Nothing Or rowNo < 2 Or rowNo > ReviewLast(ws) Then Exit Function
    If reviewManagementIdColumn < 1 Then Exit Function
    If Not ReviewNormalizeManagementId(ws.Cells(rowNo, reviewManagementIdColumn).Value2, normalizedId) Then Exit Function
    If Not reviewManagementIdCounts.Exists(normalizedId) Then Exit Function
    If CLng(reviewManagementIdCounts(normalizedId)) <> 1 Then Exit Function
    ReviewManagementRowId = normalizedId
End Function

Private Function ReviewManagementIdSnapshot(ByVal ws As Worksheet, ByVal rows As Collection) As String
    Dim rowValue As Variant, managementId As String
    For Each rowValue In rows
        managementId = ReviewManagementRowId(ws, CLng(rowValue))
        ReviewManagementIdSnapshot = ReviewManagementIdSnapshot & CStr(rowValue) & "|" & _
            CStr(Len(managementId)) & ":" & managementId & ";"
    Next rowValue
End Function

Private Sub ReviewAppendUnique(ByVal rows As Collection, ByVal rowNo As Long)
    Dim v As Variant
    For Each v In rows
        If CLng(v) = rowNo Then Exit Sub
    Next v
    rows.Add rowNo
End Sub

Private Function ReviewSourceStableKey(ByVal src As Worksheet, ByVal srcRow As Long, ByVal yearValue As Long, _
    ByVal csvIds As Object, ByVal csvYearBars As Object, ByVal csvYearNumbers As Object) As String
    Dim fileId As String, barcode As String, fileNumber As String
    If srcRow < 2 Or ReviewFiscal(src, srcRow) <> yearValue Then Exit Function
    fileId = ReviewField(src, srcRow, "ファイルID")
    barcode = ReviewField(src, srcRow, "バーコード番号")
    fileNumber = ReviewPadding(ReviewField(src, srcRow, "ファイル番号"))
    If Len(fileId) = 0 Or Len(fileNumber) = 0 Or Len(barcode) = 0 Then Exit Function
    If ReviewRows(csvIds, fileId).Count <> 1 Then Exit Function
    If ReviewRows(csvYearBars, ReviewYearIndexKey(yearValue, barcode)).Count <> 1 Then Exit Function
    If ReviewRows(csvYearNumbers, ReviewYearIndexKey(yearValue, fileNumber)).Count <> 1 Then Exit Function
    ReviewSourceStableKey = "SR1|" & CStr(yearValue) & "|" & _
        CStr(Len(fileId)) & ":" & fileId & "|" & _
        CStr(Len(fileNumber)) & ":" & fileNumber & "|" & _
        CStr(Len(barcode)) & ":" & barcode
End Function

Private Function ReviewFindUniqueFormalMatch(ByVal src As Worksheet, ByVal srcRow As Long, _
    ByVal dst As Worksheet, ByVal yearValue As Long, ByVal idIndex As Object, ByVal csvIds As Object, _
    ByVal csvYearBars As Object, ByVal csvYearNumbers As Object, ByVal sourceStableKey As String, _
    ByRef matchedRow As Long) As Boolean
    Dim fileId As String, barcode As String, fileNumber As String, idRows As Collection
    matchedRow = 0
    If Len(sourceStableKey) = 0 Then Exit Function
    fileId = ReviewField(src, srcRow, "ファイルID")
    barcode = ReviewField(src, srcRow, "バーコード番号")
    fileNumber = ReviewPadding(ReviewField(src, srcRow, "ファイル番号"))
    If Len(fileId) = 0 Or Len(barcode) = 0 Or Len(fileNumber) = 0 Then Exit Function
    If ReviewRows(csvIds, fileId).Count <> 1 Then Exit Function
    If ReviewRows(csvYearBars, ReviewYearIndexKey(yearValue, barcode)).Count <> 1 Then Exit Function
    If ReviewRows(csvYearNumbers, ReviewYearIndexKey(yearValue, fileNumber)).Count <> 1 Then Exit Function
    Set idRows = ReviewRows(idIndex, fileId)
    If idRows.Count <> 1 Then Exit Function
    matchedRow = CLng(idRows(1))
    If ReviewFiscal(dst, matchedRow) <> yearValue Then Exit Function
    If ReviewField(dst, matchedRow, "ファイルID") <> fileId Then Exit Function
    If ReviewPadding(ReviewField(dst, matchedRow, "ファイル番号")) <> fileNumber Then Exit Function
    If ReviewField(dst, matchedRow, "バーコード番号") <> barcode Then Exit Function
    If ReviewField(dst, matchedRow, "同期状態") <> "同期済" Then Exit Function
    ReviewFindUniqueFormalMatch = True
End Function

Private Function ReviewExactFolderPlannedRows(ByVal src As Worksheet, ByVal srcRow As Long, _
    ByVal dst As Worksheet, ByVal folderIndex As Object, ByVal yearValue As Long) As Collection
    Dim exactRows As New Collection, rowValue As Variant, folderRows As Collection
    Dim sourceFolder As String, sourceName As String
    sourceName = ReviewName(src, srcRow, True, True)
    If Len(sourceName) = 0 Then
        Set ReviewExactFolderPlannedRows = exactRows
        Exit Function
    End If
    sourceFolder = ReviewTextKey(ReviewValue(src, srcRow, ReviewCol(src, "タイトル", "個別フォルダー")))
    Set folderRows = ReviewRows(folderIndex, sourceFolder)
    For Each rowValue In folderRows
        If ReviewFiscal(dst, CLng(rowValue)) = yearValue And _
           ReviewName(dst, CLng(rowValue), False, True) = sourceName And _
           ReviewField(dst, CLng(rowValue), "同期状態") = "登録予定" And _
           Len(ReviewField(dst, CLng(rowValue), "ファイルID")) = 0 Then ReviewAppendUnique exactRows, CLng(rowValue)
    Next rowValue
    Set ReviewExactFolderPlannedRows = exactRows
End Function

Private Function ReviewExactFolderAssignedRows(ByVal src As Worksheet, ByVal srcRow As Long, _
    ByVal dst As Worksheet, ByVal folderIndex As Object, ByVal yearValue As Long, _
    ByVal sourceFileId As String) As Collection
    Dim conflictRows As New Collection, rowValue As Variant, folderRows As Collection
    Dim sourceFolder As String, sourceName As String, rowFileId As String
    sourceName = ReviewName(src, srcRow, True, True)
    If Len(sourceName) = 0 Then
        Set ReviewExactFolderAssignedRows = conflictRows
        Exit Function
    End If
    sourceFolder = ReviewTextKey(ReviewValue(src, srcRow, ReviewCol(src, "タイトル", "個別フォルダー")))
    Set folderRows = ReviewRows(folderIndex, sourceFolder)
    For Each rowValue In folderRows
        If ReviewFiscal(dst, CLng(rowValue)) = yearValue And _
           ReviewName(dst, CLng(rowValue), False, True) = sourceName And _
           ReviewField(dst, CLng(rowValue), "同期状態") = "登録予定" Then
            rowFileId = ReviewField(dst, CLng(rowValue), "ファイルID")
            If Len(rowFileId) > 0 And rowFileId <> sourceFileId Then ReviewAppendUnique conflictRows, CLng(rowValue)
        End If
    Next rowValue
    Set ReviewExactFolderAssignedRows = conflictRows
End Function

Private Function ReviewCandidateRows(ByVal src As Worksheet, ByVal sRow As Long, ByVal dst As Worksheet, _
    ByVal idIndex As Object, ByVal barIndex As Object, ByVal noIndex As Object, ByVal nameIndex As Object, _
    ByVal guideIndex As Object) As Collection
    Dim rows As New Collection, v As Variant, id As String, bar As String, no As String, name As String, guide As String, sourceFolder As String
    id = ReviewField(src, sRow, "ファイルID")
    bar = ReviewField(src, sRow, "バーコード番号")
    no = ReviewPadding(ReviewField(src, sRow, "ファイル番号"))
    name = ReviewName(src, sRow, True, True)
    guide = ReviewName(src, sRow, True, False)
    sourceFolder = ReviewValue(src, sRow, ReviewCol(src, "タイトル", "個別フォルダー"))
    For Each v In ReviewRows(idIndex, id): ReviewAppendUnique rows, CLng(v): Next v
    For Each v In ReviewRows(barIndex, bar): ReviewAppendUnique rows, CLng(v): Next v
    For Each v In ReviewRows(noIndex, no): ReviewAppendUnique rows, CLng(v): Next v
    For Each v In ReviewRows(nameIndex, name)
        If ReviewField(dst, CLng(v), "同期状態") = "登録予定" Then ReviewAppendUnique rows, CLng(v)
    Next v
    If Len(guide) > 0 Then
        For Each v In ReviewRows(guideIndex, guide)
            If ReviewField(dst, CLng(v), "同期状態") = "登録予定" And _
                Len(ReviewField(dst, CLng(v), "ファイルID")) = 0 And _
                ReviewSimilar(sourceFolder, ReviewValue(dst, CLng(v), ReviewCol(dst, "個別フォルダー", "タイトル"))) Then _
                    ReviewAppendUnique rows, CLng(v)
        Next v
    End If
    Set ReviewCandidateRows = rows
End Function

Public Function SyncReviewCandidates(ByVal yearValue As Long) As Collection
    Dim src As Worksheet, dst As Worksheet, cases As New Collection
    Dim ids As Object, bars As Object, numbers As Object, names As Object, guides As Object, folders As Object, csvIds As Object
    Dim csvBars As Object, csvNumbers As Object, csvNames As Object, csvFolders As Object, targetCount As Long, dRow As Long, sRow As Long
    Dim csvYearBars As Object, csvYearNumbers As Object, sourceFiscalYear As Long
    Dim srcLast As Long, dstLast As Long, destRows As Collection, idRows As Collection, plannedRows As Collection, v As Variant
    Dim fid As String, bar As String, no As String, name As String, guide As String
    Dim reason As String, collision As Boolean, planned As Boolean, chosen As Collection
    Dim sourceStableKey As String, matchedFormalRow As Long, assignedConflictRows As Collection
    If yearValue < 1900 Or yearValue > 2200 Then Err.Raise vbObjectError + 6104, , "対象年度が不正です。"
    Set reviewHeaderCache = CreateObject("Scripting.Dictionary")
    Set reviewLastCache = CreateObject("Scripting.Dictionary")
    Set src = ThisWorkbook.Worksheets(CSV_SHEET)
    Set dst = ThisWorkbook.Worksheets(DEST_SHEET)
    If ReviewCol(src, "ファイルID") = 0 Or ReviewCol(src, "ファイル番号") = 0 Or _
        ReviewCol(src, "バーコード番号") = 0 Or ReviewCol(dst, "同期状態") = 0 Or _
        ReviewCol(dst, "ファイルID") = 0 Or ReviewCol(dst, "ファイル番号") = 0 Or _
        ReviewCol(dst, "バーコード番号") = 0 Then Err.Raise vbObjectError + 6105, , "必要な見出しが不足しています。"
    Set ids = CreateObject("Scripting.Dictionary")
    Set bars = CreateObject("Scripting.Dictionary")
    Set numbers = CreateObject("Scripting.Dictionary")
    Set names = CreateObject("Scripting.Dictionary")
    Set guides = CreateObject("Scripting.Dictionary")
    Set folders = CreateObject("Scripting.Dictionary")
    Set csvIds = CreateObject("Scripting.Dictionary")
    Set csvBars = CreateObject("Scripting.Dictionary")
    Set csvNumbers = CreateObject("Scripting.Dictionary")
    Set csvNames = CreateObject("Scripting.Dictionary")
    Set csvFolders = CreateObject("Scripting.Dictionary")
    Set csvYearBars = CreateObject("Scripting.Dictionary")
    Set csvYearNumbers = CreateObject("Scripting.Dictionary")
    srcLast = ReviewLast(src)
    dstLast = ReviewLast(dst)
    ReviewBuildManagementIdIndex dst
    For dRow = 2 To dstLast
        ReviewMapAdd ids, ReviewField(dst, dRow, "ファイルID"), dRow
        ReviewMapAdd bars, ReviewField(dst, dRow, "バーコード番号"), dRow
        ReviewMapAdd numbers, ReviewPadding(ReviewField(dst, dRow, "ファイル番号")), dRow
        ReviewMapAdd names, ReviewName(dst, dRow, False, True), dRow
        ReviewMapAdd guides, ReviewName(dst, dRow, False, False), dRow
        ReviewMapAdd folders, ReviewTextKey(ReviewValue(dst, dRow, ReviewCol(dst, "個別フォルダー", "個別フォルダ", "タイトル"))), dRow
    Next dRow
    For sRow = 2 To srcLast
        fid = ReviewField(src, sRow, "ファイルID")
        bar = ReviewField(src, sRow, "バーコード番号")
        no = ReviewPadding(ReviewField(src, sRow, "ファイル番号"))
        sourceFiscalYear = ReviewFiscal(src, sRow)
        ReviewMapAdd csvIds, fid, sRow
        ReviewMapAdd csvBars, bar, sRow
        ReviewMapAdd csvNumbers, no, sRow
        ReviewMapAdd csvNames, ReviewName(src, sRow, True, True), sRow
        ReviewMapAdd csvFolders, ReviewTextKey(ReviewValue(src, sRow, ReviewCol(src, "タイトル", "個別フォルダー"))), sRow
        If sourceFiscalYear > 0 Then
            If Len(bar) > 0 Then ReviewMapAdd csvYearBars, ReviewYearIndexKey(sourceFiscalYear, bar), sRow
            If Len(no) > 0 Then ReviewMapAdd csvYearNumbers, ReviewYearIndexKey(sourceFiscalYear, no), sRow
        End If
        If sourceFiscalYear = yearValue Then targetCount = targetCount + 1
    Next sRow
    If targetCount = 0 Then Err.Raise vbObjectError + 6136, , "対象年度の目録CSVが0行です。最新CSVを取り込んでください。"
    For sRow = 2 To srcLast
        If ReviewFiscal(src, sRow) <> yearValue Then GoTo NextSource
        fid = ReviewField(src, sRow, "ファイルID")
        bar = ReviewField(src, sRow, "バーコード番号")
        no = ReviewPadding(ReviewField(src, sRow, "ファイル番号"))
        name = ReviewName(src, sRow, True, True)
        guide = ReviewName(src, sRow, True, False)
        Set idRows = ReviewRows(ids, fid)
        Set destRows = ReviewCandidateRows(src, sRow, dst, ids, bars, numbers, names, guides)
        collision = False: planned = False: reason = "": sourceStableKey = "": matchedFormalRow = 0
        If Len(fid) = 0 Then
            collision = True: reason = "CSVに正式ファイルIDがありません。"
        ElseIf ReviewRows(csvIds, fid).Count > 1 Or idRows.Count > 1 Then
            collision = True: reason = "正式IDがCSVまたはExcelに複数あります。自動統合禁止。"
        ElseIf Len(bar) > 0 And ReviewRows(csvYearBars, ReviewYearIndexKey(yearValue, bar)).Count > 1 Then
            collision = True: reason = "対象年度のCSVでバーコード番号が重複しています。正式レコードを一意に特定できません。"
        ElseIf Len(no) > 0 And ReviewRows(csvYearNumbers, ReviewYearIndexKey(yearValue, no)).Count > 1 Then
            collision = True: reason = "対象年度のCSVでファイル番号が重複しています。正式レコードを一意に特定できません。"
        ElseIf Len(bar) > 0 And ReviewRows(bars, bar).Count > 0 Then
            For Each v In ReviewRows(bars, bar)
                If ReviewField(dst, CLng(v), "ファイルID") <> fid Then collision = True
            Next v
            If collision Then reason = "別のExcel行とバーコードが競合します。"
        End If
        If Not collision And Len(no) > 0 Then
            For Each v In ReviewRows(numbers, no)
                If ReviewField(dst, CLng(v), "ファイルID") <> fid Then collision = True
            Next v
            If collision Then reason = "別のExcel行とファイル番号が競合します。"
        End If
        If Not collision And idRows.Count = 1 Then
            matchedFormalRow = CLng(idRows(1))
            If ReviewFiscal(dst, matchedFormalRow) <> yearValue Then
                collision = True: reason = "正式ファイルIDの一致行が別年度にあります。"
            ElseIf Len(no) > 0 And ReviewPadding(ReviewField(dst, matchedFormalRow, "ファイル番号")) <> no Then
                collision = True: reason = "正式ファイルID一致行のファイル番号がCSVと一致しません。"
            ElseIf Len(bar) > 0 And ReviewField(dst, matchedFormalRow, "バーコード番号") <> bar Then
                collision = True: reason = "正式ファイルID一致行のバーコード番号がCSVと一致しません。"
            ElseIf ReviewField(dst, matchedFormalRow, "同期状態") <> "同期済" Then
                collision = True: reason = "正式ファイルID一致行が同期済み状態ではありません。"
            End If
        End If
        If Not collision Then
            sourceStableKey = ReviewSourceStableKey(src, sRow, yearValue, csvIds, csvYearBars, csvYearNumbers)
            If ReviewFindUniqueFormalMatch(src, sRow, dst, yearValue, ids, csvIds, _
                csvYearBars, csvYearNumbers, sourceStableKey, matchedFormalRow) Then
                Set assignedConflictRows = ReviewExactFolderAssignedRows(src, sRow, dst, folders, yearValue, fid)
                If assignedConflictRows.Count > 0 Then _
                    cases.Add NewCase("A", yearValue, src, sRow, dst, assignedConflictRows, _
                        "正式IDの一致先とは別に、作成年度・第1/第2ガイド・個別フォルダー名が一致する登録予定行へ別の正式IDがあります。統合せず確認してください。")
                Set plannedRows = ReviewExactFolderPlannedRows(src, sRow, dst, folders, yearValue)
                If plannedRows.Count > 0 Then
                    reason = "正式ID・年度・ファイル番号・バーコードが一意一致するため弱い類似候補は除外しました。作成年度・第1/第2ガイド・個別フォルダー名が正規化一致する登録予定行は独立候補として確認します。"
                    cases.Add NewCase("B", yearValue, src, sRow, dst, plannedRows, reason, sourceStableKey)
                End If
            Else
                For Each v In destRows
                    If ReviewField(dst, CLng(v), "同期状態") = "登録予定" And _
                        Len(ReviewField(dst, CLng(v), "ファイルID")) = 0 Then planned = True
                Next v
                If planned Then
                    reason = "同名または同一作成年度・第1/第2ガイドの類似候補です。名前だけで統合しません。"
                    cases.Add NewCase("B", yearValue, src, sRow, dst, destRows, reason, sourceStableKey)
                End If
            End If
        Else
            cases.Add NewCase("A", yearValue, src, sRow, dst, destRows, reason)
        End If
NextSource:
    Next sRow
    For dRow = 2 To dstLast
        If ReviewFiscal(dst, dRow) <> yearValue Then GoTo NextDestination
        If ReviewField(dst, dRow, "同期状態") <> "削除予定" Then GoTo NextDestination
        fid = ReviewField(dst, dRow, "ファイルID")
        bar = ReviewField(dst, dRow, "バーコード番号")
        no = ReviewPadding(ReviewField(dst, dRow, "ファイル番号"))
        name = ReviewName(dst, dRow, False, True)
        If Len(name) = 0 Or _
           ReviewRows(csvIds, fid).Count > 0 Or ReviewRows(csvBars, bar).Count > 0 Or _
           ReviewRows(csvNumbers, no).Count > 0 Or ReviewRows(csvNames, name).Count > 0 Or _
           ReviewRows(csvFolders, ReviewTextKey(ReviewValue(dst, dRow, ReviewCol(dst, "個別フォルダー", "タイトル")))).Count > 0 Then GoTo NextDestination
        If Len(fid) > 0 Then
            If ReviewRows(ids, fid).Count > 1 Then GoTo NextDestination
        End If
        Set chosen = New Collection
        chosen.Add dRow
        cases.Add NewCase("C", yearValue, src, 0, dst, chosen, "CSVに同一ID・バーコード・番号・名称の候補がありません。削除前に再検証します。")
NextDestination:
    Next dRow
    Set SyncReviewCandidates = cases
End Function

Private Function ReviewOtherRowsMatch(ByVal index As Object, ByVal key As String, ByVal excludedRow As Long) As Boolean
    Dim rows As Collection
    If Len(key) = 0 Then Exit Function
    Set rows = ReviewRows(index, key)
    If rows.Count = 0 Then Exit Function
    ReviewOtherRowsMatch = (rows.Count > 1 Or CLng(rows(1)) <> excludedRow)
End Function

' The three index lists are already in worksheet order. Merge them without
' re-scanning the worksheet, retaining one copy of each matching row.
Private Function ReviewMatchingRows(ByVal ids As Object, ByVal bars As Object, ByVal numbers As Object, _
    ByVal fid As String, ByVal barcode As String, ByVal fileNo As String) As Collection
    Dim lists(0 To 2) As Collection, positions(0 To 2) As Long
    Dim result As New Collection, i As Long, nextRow As Long, candidate As Long
    Set lists(0) = ReviewRows(ids, fid)
    Set lists(1) = ReviewRows(bars, barcode)
    Set lists(2) = ReviewRows(numbers, fileNo)
    For i = 0 To 2: positions(i) = 1: Next i
    Do
        nextRow = 0
        For i = 0 To 2
            If positions(i) <= lists(i).Count Then
                candidate = CLng(lists(i)(positions(i)))
                If nextRow = 0 Or candidate < nextRow Then nextRow = candidate
            End If
        Next i
        If nextRow = 0 Then Exit Do
        result.Add nextRow
        For i = 0 To 2
            If positions(i) <= lists(i).Count Then
                If CLng(lists(i)(positions(i))) = nextRow Then positions(i) = positions(i) + 1
            End If
        Next i
    Loop
    Set ReviewMatchingRows = result
End Function

Public Function SyncReviewCopyIdCandidates(ByVal yearValue As Long) As Collection
    Dim src As Worksheet, dst As Worksheet, cases As New Collection
    Dim sRow As Long, dRow As Long, srcLast As Long, dstLast As Long
    Dim fid As String, fileNo As String, barcode As String, sourceName As String
    Dim keeperRows As Collection, collisionRows As Collection, candidates As Collection
    Dim v As Variant, rowNo As Long, valid As Boolean, reason As String
    Dim allRows As Collection, copyCase As Object, datasetSignature As String
    Dim csvIds As Object, csvBars As Object, csvTargetNumbers As Object
    Dim dstIds As Object, dstBars As Object, dstTargetNumbers As Object

    If yearValue < 1900 Or yearValue > 2200 Then Err.Raise vbObjectError + 6140, , "対象年度が不正です。"
    Set reviewHeaderCache = CreateObject("Scripting.Dictionary")
    Set reviewLastCache = CreateObject("Scripting.Dictionary")
    Set src = ThisWorkbook.Worksheets(CSV_SHEET)
    Set dst = ThisWorkbook.Worksheets(DEST_SHEET)
    If ReviewCol(src, "ファイルID") = 0 Or ReviewCol(src, "ファイル番号") = 0 Or _
       ReviewCol(src, "バーコード番号") = 0 Or ReviewCol(src, "年度（和暦）", "年度(和暦)", "年度") = 0 Or _
       ReviewCol(src, "作成年度（和暦）", "作成年度(和暦)") = 0 Or _
       ReviewCol(src, "分類名２", "分類名2", "第1ガイド", "第１ガイド") = 0 Or _
       ReviewCol(src, "分類名３", "分類名3", "第2ガイド", "第２ガイド") = 0 Or _
       ReviewCol(src, "タイトル", "個別フォルダー", "個別フォルダ") = 0 Or _
       ReviewCol(dst, "ファイルID") = 0 Or ReviewCol(dst, "ファイル番号") = 0 Or _
       ReviewCol(dst, "バーコード番号") = 0 Or ReviewCol(dst, "同期状態") = 0 Or _
       ReviewCol(dst, "年度（和暦）", "年度(和暦)", "年度") = 0 Or _
       ReviewCol(dst, "作成年度（和暦）", "作成年度(和暦)") = 0 Or _
       ReviewCol(dst, "第1ガイド", "第１ガイド", "分類名２") = 0 Or _
       ReviewCol(dst, "第2ガイド", "第２ガイド", "分類名３") = 0 Or _
       ReviewCol(dst, "個別フォルダー", "個別フォルダ", "タイトル") = 0 Then _
        Err.Raise vbObjectError + 6141, , "D候補判定に必要な見出しが不足または曖昧です。"

    srcLast = ReviewLast(src): dstLast = ReviewLast(dst)
    datasetSignature = ReviewCopyDatasetSignature(src, dst)
    Set csvIds = CreateObject("Scripting.Dictionary")
    Set csvBars = CreateObject("Scripting.Dictionary")
    Set csvTargetNumbers = CreateObject("Scripting.Dictionary")
    Set dstIds = CreateObject("Scripting.Dictionary")
    Set dstBars = CreateObject("Scripting.Dictionary")
    Set dstTargetNumbers = CreateObject("Scripting.Dictionary")
    ' IDs and barcodes remain global safety keys; file numbers are scoped to the selected fiscal year.
    For sRow = 2 To srcLast
        ReviewMapAdd csvIds, ReviewField(src, sRow, "ファイルID"), sRow
        ReviewMapAdd csvBars, ReviewField(src, sRow, "バーコード番号"), sRow
        If ReviewFiscal(src, sRow) = yearValue Then ReviewMapAdd csvTargetNumbers, ReviewPadding(ReviewField(src, sRow, "ファイル番号")), sRow
    Next sRow
    For dRow = 2 To dstLast
        ReviewMapAdd dstIds, ReviewField(dst, dRow, "ファイルID"), dRow
        ReviewMapAdd dstBars, ReviewField(dst, dRow, "バーコード番号"), dRow
        If ReviewFiscal(dst, dRow) = yearValue Then ReviewMapAdd dstTargetNumbers, ReviewPadding(ReviewField(dst, dRow, "ファイル番号")), dRow
    Next dRow
    For sRow = 2 To srcLast
        If ReviewFiscal(src, sRow) <> yearValue Then GoTo NextCopySource
        fid = ReviewField(src, sRow, "ファイルID")
        fileNo = ReviewPadding(ReviewField(src, sRow, "ファイル番号"))
        barcode = ReviewField(src, sRow, "バーコード番号")
        sourceName = ReviewName(src, sRow, True, True)
        If Len(fid) = 0 Or Len(fileNo) = 0 Or Len(barcode) = 0 Or Len(sourceName) = 0 Then GoTo NextCopySource

        If ReviewOtherRowsMatch(csvIds, fid, sRow) Or _
           ReviewOtherRowsMatch(csvBars, barcode, sRow) Or _
           ReviewOtherRowsMatch(csvTargetNumbers, fileNo, sRow) Then GoTo NextCopySource

        Set keeperRows = New Collection: Set candidates = New Collection
        Set collisionRows = ReviewMatchingRows(dstIds, dstBars, dstTargetNumbers, fid, barcode, fileNo)
        For Each v In collisionRows
            dRow = CLng(v)
            If ReviewField(dst, dRow, "ファイルID") = fid And _
               ReviewPadding(ReviewField(dst, dRow, "ファイル番号")) = fileNo And _
               ReviewField(dst, dRow, "バーコード番号") = barcode And _
               ReviewFiscal(dst, dRow) = yearValue And _
               ReviewName(dst, dRow, False, True) = sourceName Then keeperRows.Add dRow
        Next v
        If keeperRows.Count <> 1 Then GoTo NextCopySource
        rowNo = CLng(keeperRows(1))
        If ReviewField(dst, rowNo, "同期状態") <> "登録予定" And _
           ReviewField(dst, rowNo, "同期状態") <> "同期済" Then GoTo NextCopySource
        If ReviewName(dst, rowNo, False, False) <> ReviewName(src, sRow, True, False) Then GoTo NextCopySource

        valid = True
        For Each v In collisionRows
            dRow = CLng(v)
            If dRow <> rowNo Then
                If ReviewFiscal(dst, dRow) <> yearValue Then
                    valid = False
                Else
                    If ReviewField(dst, dRow, "同期状態") <> "登録予定" Then valid = False
                    If Len(ReviewName(dst, dRow, False, True)) = 0 Or _
                       Len(ReviewName(dst, dRow, False, False)) = 0 Then valid = False
                    If ReviewName(dst, dRow, False, True) = sourceName Then valid = False
                    If ReviewOtherRowsMatch(csvIds, ReviewField(dst, dRow, "ファイルID"), sRow) Or _
                       ReviewOtherRowsMatch(csvBars, ReviewField(dst, dRow, "バーコード番号"), sRow) Or _
                       ReviewOtherRowsMatch(csvTargetNumbers, ReviewPadding(ReviewField(dst, dRow, "ファイル番号")), sRow) Then valid = False
                    candidates.Add dRow
                End If
            End If
        Next v
        If Not valid Or candidates.Count = 0 Then GoTo NextCopySource

        reason = "CSVと正式番号・作成年度・第1/第2ガイド・個別フォルダー名が一致するExcel行" & _
            rowNo & "を保持対象として一意に確認。別の登録予定行は番号競合候補です。" & _
            "削除・クリアは行わず、管理システムの登録状況を人が確認してください。"
        Set allRows = New Collection
        allRows.Add rowNo
        For Each v In candidates: allRows.Add CLng(v): Next v
        Set copyCase = NewCase("D", yearValue, src, sRow, dst, allRows, reason)
        copyCase.Add "copyKeeperRow", rowNo
        copyCase.Add "copyCandidateRows", candidates
        copyCase.Add "copyDatasetSignature", datasetSignature
        cases.Add copyCase
NextCopySource:
    Next sRow
    Set SyncReviewCopyIdCandidates = cases
End Function

Private Function ReviewCopySheetSignature(ByVal ws As Worksheet) As String
    Dim lastRow As Long, lastCol As Long, col As Long, rowNo As Long
    Dim headerText As String, valueText As String, parts() As String
    lastRow = ReviewLast(ws)
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For col = 1 To lastCol
        If IsError(ws.Cells(1, col).Value2) Then Err.Raise vbObjectError + 6142, , "見出しにエラー値があります。"
        If Not ReviewIsInternalMetadataHeader(ws.Cells(1, col).Value2) Then
            valueText = TypeName(ws.Cells(1, col).Value2) & ":" & CStr(ws.Cells(1, col).Value2)
            If ws.Cells(1, col).HasFormula Then valueText = valueText & "#" & CStr(ws.Cells(1, col).Formula)
            headerText = headerText & CStr(Len(valueText)) & ":" & valueText
        End If
    Next col
    ReDim parts(0 To lastRow - 1)
    parts(0) = ws.Name & "|" & ReviewDigest(headerText)
    For rowNo = 2 To lastRow
        parts(rowNo - 1) = ReviewDigest(ReviewSignature(ws, rowNo))
    Next rowNo
    ReviewCopySheetSignature = ReviewDigest(Join(parts, "|"))
End Function

Private Function ReviewCopyDatasetSignature(ByVal src As Worksheet, ByVal dst As Worksheet) As String
    ReviewCopyDatasetSignature = ReviewDigest(ReviewCopySheetSignature(src) & "|" & ReviewCopySheetSignature(dst))
End Function

Private Function ReviewCopyCollectionHasRow(ByVal rows As Collection, ByVal rowNo As Long) As Boolean
    Dim value As Variant
    If rows Is Nothing Then Exit Function
    For Each value In rows
        If CLng(value) = rowNo Then ReviewCopyCollectionHasRow = True: Exit Function
    Next value
End Function

Private Function ReviewCopyCaseCurrent(ByVal expected As Object, ByVal selectedRow As Long) As Boolean
    Dim current As Collection, item As Object, i As Long
    If expected Is Nothing Then Exit Function
    If Not expected.Exists("copyKeeperRow") Or Not expected.Exists("copyCandidateRows") Or _
       Not expected.Exists("copyDatasetSignature") Then Exit Function
    If CLng(expected("copyKeeperRow")) = selectedRow Then Exit Function
    If Not ReviewCopyCollectionHasRow(expected("copyCandidateRows"), selectedRow) Then Exit Function
    Set current = SyncReviewCopyIdCandidates(CLng(expected("year")))
    For i = 1 To current.Count
        Set item = current(i)
        If CStr(item("key")) = CStr(expected("key")) And _
           CStr(item("sourceSignature")) = CStr(expected("sourceSignature")) And _
           CStr(item("destinationSignature")) = CStr(expected("destinationSignature")) And _
           CLng(item("copyKeeperRow")) = CLng(expected("copyKeeperRow")) And _
           CStr(item("copyDatasetSignature")) = CStr(expected("copyDatasetSignature")) And _
           ReviewCopyCollectionHasRow(item("copyCandidateRows"), selectedRow) Then
            ReviewCopyCaseCurrent = True
            Exit Function
        End If
    Next i
End Function

Private Function ReviewCopyAuditNextRow(ByVal ws As Worksheet) As Long
    Dim lastCell As Range
    Set lastCell = ws.Cells.Find(What:="*", After:=ws.Cells(1, 1), LookIn:=xlFormulas, _
        LookAt:=xlPart, SearchOrder:=xlByRows, SearchDirection:=xlPrevious, SearchFormat:=False)
    If lastCell Is Nothing Then
        ReviewCopyAuditNextRow = 2
    Else
        ReviewCopyAuditNextRow = Application.Max(2, lastCell.Row + 1)
    End If
End Function

Private Sub ReviewPreflightCopyAudit()
    Dim ws As Worksheet, r As Long, col As Long
    If ThisWorkbook.ReadOnly Then Err.Raise vbObjectError + 6156, , "監査履歴を書けません: ブックが読み取り専用です。"
    Set ws = ReviewAudit(False)
    If ws Is Nothing Then Err.Raise vbObjectError + 6156, , _
        "同期候補確認履歴シートがありません。先に安全なコピーで履歴シートを準備してください。番号クリアは中止します。"
    If ws.ProtectContents Then Err.Raise vbObjectError + 6156, , "監査履歴シートが保護されています。番号クリアは中止します。"
    r = ReviewCopyAuditNextRow(ws)
    If r > ws.Rows.Count Then Err.Raise vbObjectError + 6156, , "監査履歴シートに追記する行がありません。"
    For col = 1 To 12
        If ws.Cells(r, col).MergeCells Then _
            Err.Raise vbObjectError + 6156, , "監査履歴の追記先に結合セルがあります。番号クリアは中止します。"
    Next col
End Sub

Private Sub ReviewWriteCopyClearAudit(ByVal item As Object, ByVal decision As String, _
    ByVal detail As String, ByVal backup As String)
    Dim ws As Worksheet, r As Long
    Set ws = ReviewAudit(False)
    If ws Is Nothing Then Err.Raise vbObjectError + 6156, , "監査履歴シートがありません。"
    r = ReviewCopyAuditNextRow(ws)
    ws.Cells(r, 1).Value2 = Now
    ws.Cells(r, 2).Value2 = CLng(item("year"))
    ws.Cells(r, 3).Value2 = "D"
    ws.Cells(r, 4).NumberFormat = "@"
    ws.Cells(r, 9).NumberFormat = "@": ws.Cells(r, 10).NumberFormat = "@"
    ws.Cells(r, 4).Value2 = CStr(item("key"))
    ws.Cells(r, 6).Value2 = CLng(item("sourceRow"))
    ws.Cells(r, 7).Value2 = CStr(item("rows"))
    ws.Cells(r, 8).Value2 = CStr(item("reason"))
    ws.Cells(r, 9).Value2 = ReviewDigest(CStr(item("sourceSignature")))
    ws.Cells(r, 10).Value2 = ReviewDigest(CStr(item("destinationSignature")))
    ws.Cells(r, 11).Value2 = detail
    ws.Cells(r, 12).Value2 = backup
    ws.Cells(r, 5).Value2 = decision
End Sub

Private Function ReviewRestoreCopyIdCells(ByVal ws As Worksheet, ByVal rowNo As Long, _
    ByRef cols() As Long, ByRef oldValues() As Variant, ByRef oldFormulas() As String, _
    ByRef wasFormula() As Boolean, ByRef oldFormats() As String) As Boolean
    Dim i As Long, restoreOk As Boolean
    restoreOk = True
    On Error Resume Next
    For i = 1 To 3
        Err.Clear
        If wasFormula(i) Then
            ws.Cells(rowNo, cols(i)).Formula = oldFormulas(i)
        Else
            ws.Cells(rowNo, cols(i)).Value2 = oldValues(i)
        End If
        If Err.Number <> 0 Then restoreOk = False
        Err.Clear
        ws.Cells(rowNo, cols(i)).NumberFormat = oldFormats(i)
        If Err.Number <> 0 Then restoreOk = False
        Err.Clear
        If ws.Cells(rowNo, cols(i)).HasFormula <> wasFormula(i) Then restoreOk = False
        If wasFormula(i) Then
            If CStr(ws.Cells(rowNo, cols(i)).Formula) <> oldFormulas(i) Then restoreOk = False
        ElseIf IsEmpty(oldValues(i)) Then
            If Not IsEmpty(ws.Cells(rowNo, cols(i)).Value2) Then restoreOk = False
        Else
            If CStr(ws.Cells(rowNo, cols(i)).Value2) <> CStr(oldValues(i)) Then restoreOk = False
        End If
        If CStr(ws.Cells(rowNo, cols(i)).NumberFormat) <> oldFormats(i) Then restoreOk = False
    Next i
    On Error GoTo 0
    ReviewRestoreCopyIdCells = restoreOk
End Function

Public Function SyncReviewClearCopyIdentifiers(ByVal item As Object, ByVal selectedRow As Long) As Boolean
    Dim src As Worksheet, dst As Worksheet, sRow As Long, keeperRow As Long
    Dim candidateRows As Collection, cols(1 To 3) As Long, oldValues(1 To 3) As Variant
    Dim oldFormulas(1 To 3) As String, wasFormula(1 To 3) As Boolean, oldFormats(1 To 3) As String
    Dim i As Long, backup As String, details As String, failureText As String
    Dim prevEvents As Boolean, eventsCaptured As Boolean, changing As Boolean, restored As Boolean
    Dim criticalFailure As Boolean, auditStarted As Boolean
    On Error GoTo Failed
    If item Is Nothing Then Err.Raise vbObjectError + 6143, , "D候補を指定してください。"
    If CStr(item("category")) <> "D" Or CLng(item("sourceRow")) < 2 Then _
        Err.Raise vbObjectError + 6144, , "D候補以外は番号クリアできません。"
    If Not item.Exists("copyKeeperRow") Or Not item.Exists("copyCandidateRows") Or _
       Not item.Exists("copyDatasetSignature") Then Err.Raise vbObjectError + 6145, , "保持行と候補行の区分情報がありません。"
    keeperRow = CLng(item("copyKeeperRow"))
    Set candidateRows = item("copyCandidateRows")
    If selectedRow < 2 Or selectedRow = keeperRow Or _
       Not ReviewCopyCollectionHasRow(candidateRows, selectedRow) Then _
        Err.Raise vbObjectError + 6146, , "選択行はこのD案件のクリア候補ではありません。"
    Set src = ThisWorkbook.Worksheets(CSV_SHEET): Set dst = ThisWorkbook.Worksheets(DEST_SHEET)
    If ReviewField(dst, selectedRow, "同期状態") <> "登録予定" Then _
        Err.Raise vbObjectError + 6147, , "登録予定以外の行はクリアできません。"
    If ThisWorkbook.ReadOnly Or dst.ProtectContents Then _
        Err.Raise vbObjectError + 6148, , "ブックが読み取り専用またはシートが保護されています。"
    If Not ReviewCopyCaseCurrent(item, selectedRow) Then _
        Err.Raise vbObjectError + 6149, , "D候補・保持行・CSVまたはExcelデータが変わりました。再読み込みしてください。"

    sRow = CLng(item("sourceRow"))
    details = "CSVの正対応行:" & vbCrLf & ReviewDescription(src, sRow, True) & vbCrLf & _
        "保持するExcel行:" & vbCrLf & ReviewDescription(dst, keeperRow, False) & vbCrLf & _
        "クリア候補のExcel行:" & vbCrLf & ReviewDescription(dst, selectedRow, False)
    If MsgBox(details & vbCrLf & vbCrLf & _
        "管理システムで、このクリア候補のフォルダーが未登録であることを確認してください。確認できましたか？", _
        vbYesNo + vbQuestion + vbDefaultButton2, "システム登録状況の確認") <> vbYes Then Exit Function
    If Not ReviewCopyCaseCurrent(item, selectedRow) Then _
        Err.Raise vbObjectError + 6150, , "確認中にCSVまたはExcelデータが変わりました。"
    If MsgBox(details & vbCrLf & vbCrLf & _
        "選択したExcel行のファイルID・ファイル番号・バーコード番号だけを空欄にします。" & vbCrLf & _
        "同期状態・名称・作成年度・ガイド・書式・他セル・行自体は変更しません。" & vbCrLf & _
        "この操作はStep2の正式同期ではありません。バックアップを作成して実行しますか？", _
        vbYesNo + vbExclamation + vbDefaultButton2, "番号クリアの最終確認") <> vbYes Then Exit Function
    If Not ReviewCopyCaseCurrent(item, selectedRow) Then _
        Err.Raise vbObjectError + 6151, , "最終確認中にCSVまたはExcelデータが変わりました。"

    cols(1) = ReviewCol(dst, "ファイルID")
    cols(2) = ReviewCol(dst, "ファイル番号")
    cols(3) = ReviewCol(dst, "バーコード番号")
    For i = 1 To 3
        oldValues(i) = dst.Cells(selectedRow, cols(i)).Value2
        wasFormula(i) = dst.Cells(selectedRow, cols(i)).HasFormula
        If wasFormula(i) Then oldFormulas(i) = CStr(dst.Cells(selectedRow, cols(i)).Formula)
        oldFormats(i) = CStr(dst.Cells(selectedRow, cols(i)).NumberFormat)
    Next i
    ReviewPreflightCopyAudit
    backup = ReviewBackup("番号クリア前")
    If Not ReviewCopyCaseCurrent(item, selectedRow) Then _
        Err.Raise vbObjectError + 6152, , "バックアップ作成中にCSVまたはExcelデータが変わりました。"
    ReviewPreflightCopyAudit
    If Not ReviewCopyCaseCurrent(item, selectedRow) Then _
        Err.Raise vbObjectError + 6152, , "監査履歴の確認中にCSVまたはExcelのデータが変わりました。"

    新ファイル基準表_ファイルID編集スナップショットを無効化
    prevEvents = Application.EnableEvents
    eventsCaptured = True
    Application.EnableEvents = False
    changing = True
    For i = 1 To 3: dst.Cells(selectedRow, cols(i)).ClearContents: Next i
    For i = 1 To 3
        If dst.Cells(selectedRow, cols(i)).HasFormula Or Len(CStr(dst.Cells(selectedRow, cols(i)).Value2)) > 0 Then _
            Err.Raise vbObjectError + 6153, , "識別番号セルのクリア結果を検証できません。"
        If CStr(dst.Cells(selectedRow, cols(i)).NumberFormat) <> oldFormats(i) Then _
            Err.Raise vbObjectError + 6154, , "クリア時に識別番号セルの書式が変化しました。"
    Next i
    details = "Excel行" & selectedRow & "のファイルID・ファイル番号・バーコード番号を空欄化。" & _
        "保持行Excel" & keeperRow & "および同期状態・他セルは変更していません。"
    auditStarted = True
    ReviewWriteCopyClearAudit item, "番号クリア完了", details, backup
    Application.EnableEvents = prevEvents
    changing = False
    eventsCaptured = False
    SyncReviewClearCopyIdentifiers = True
    Exit Function
Failed:
    failureText = Err.Description
    If auditStarted Then
        failureText = failureText & " / 完了履歴の追記に失敗し、履歴が一部だけ残った可能性があります。"
        criticalFailure = True
    End If
    If changing Then
        On Error Resume Next
        Application.EnableEvents = False
        restored = ReviewRestoreCopyIdCells(dst, selectedRow, cols, oldValues, oldFormulas, wasFormula, oldFormats)
        If Not restored Then
            failureText = failureText & " / 変更前の値・数式・書式を完全に復元できませんでした。"
            criticalFailure = True
        End If
        On Error GoTo 0
    End If
    If Len(backup) > 0 Then
        On Error Resume Next
        Err.Clear
        If Not eventsCaptured Then
            prevEvents = Application.EnableEvents
            eventsCaptured = (Err.Number = 0)
        End If
        If eventsCaptured Then
            Err.Clear
            Application.EnableEvents = False
        End If
        If Not eventsCaptured Or Err.Number <> 0 Then
            failureText = failureText & " / 監査時のExcelイベント抑止に失敗しました。"
            criticalFailure = True
        Else
            Err.Clear
            ReviewWriteCopyClearAudit item, "番号クリア失敗", failureText, backup
            If Err.Number <> 0 Then
                failureText = failureText & " / 失敗履歴も記録できませんでした: " & Err.Description
                criticalFailure = True
            End If
        End If
        On Error GoTo 0
    End If
    If eventsCaptured Then
        On Error Resume Next
        Err.Clear
        Application.EnableEvents = prevEvents
        If Err.Number <> 0 Then
            failureText = failureText & " / Excelイベント状態も復元できませんでした: " & Err.Description
            criticalFailure = True
        End If
        On Error GoTo 0
    End If
    If criticalFailure Then
        failureText = failureText & " ブックを保存せず終了し、事前バックアップを保全してください。"
        MsgBox "重大な復元・監査履歴失敗です: " & failureText & vbCrLf & _
            "事前バックアップ: " & backup & vbCrLf & _
            "ブックを保存せず終了してください。", vbCritical
        Err.Raise vbObjectError + 6155, "SyncReviewClearCopyIdentifiers", _
            failureText & " / 事前バックアップ: " & backup
    End If
    MsgBox "番号クリアを完了できません: " & failureText & vbCrLf & _
        "事前バックアップ: " & backup & vbCrLf & _
        "対象セルを確認し、未保存の変更がないことを確認してください。", vbCritical
End Function

Private Function ReviewDigest(ByVal text As String) As String
    Dim h1 As Double, h2 As Double, i As Long, n As Long
    h1 = 7#: h2 = 11#
    For i = 1 To Len(text)
        n = AscW(Mid$(text, i, 1))
        If n < 0 Then n = n + 65536
        h1 = h1 * 131# + n
        h1 = h1 - Fix(h1 / 2147483629#) * 2147483629#
        h2 = h2 * 137# + n
        h2 = h2 - Fix(h2 / 2147483587#) * 2147483587#
    Next i
    ReviewDigest = CStr(Len(text)) & ":" & CStr(CLng(h1)) & ":" & CStr(CLng(h2))
End Function

Private Function ReviewAudit(ByVal createIfMissing As Boolean, _
                             Optional ByVal ensureRowColumns As Boolean = False) As Worksheet
    Dim ws As Worksheet, columns As Variant, i As Long
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(REVIEW_SHEET)
    On Error GoTo 0
    If ws Is Nothing And createIfMissing Then
        Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
        ws.Name = REVIEW_SHEET
        columns = Array("時刻", "対象年度", "分類", "案件キー", "判断", "CSV行", "Excel行", "理由", "CSV署名", "Excel署名", "変更内容", "事前バックアップ")
        For i = 0 To UBound(columns): ws.Cells(1, i + 1).Value2 = columns(i): Next i
        ws.Rows(1).Font.Bold = True
    End If
    If Not ws Is Nothing And createIfMissing And ensureRowColumns Then
        Call ReviewAuditColumn(ws, "判断単位", True)
        Call ReviewAuditColumn(ws, "選択Excel行署名", True)
        Call ReviewAuditColumn(ws, "判断キー形式", True)
        Call ReviewAuditColumn(ws, "選択Excel管理行ID", True)
        Call ReviewAuditColumn(ws, "CSV安定識別キー", True)
    End If
    Set ReviewAudit = ws
End Function

Private Function ReviewAuditTextAt(ByVal ws As Worksheet, ByVal rowNo As Long, ByVal colNo As Long) As String
    If ws Is Nothing Then Exit Function
    If rowNo < 1 Or colNo < 1 Then Exit Function
    If IsError(ws.Cells(rowNo, colNo).Value2) Then Exit Function
    If IsNull(ws.Cells(rowNo, colNo).Value2) Then Exit Function
    ReviewAuditTextAt = CStr(ws.Cells(rowNo, colNo).Value2)
End Function

Private Function ReviewAuditColumn(ByVal ws As Worksheet, ByVal headerText As String, _
                                   Optional ByVal createIfMissing As Boolean = False) As Long
    Dim col As Long, lastHeaderCol As Long, foundCol As Long, foundCount As Long
    Dim lastCell As Range
    If ws Is Nothing Then Exit Function
    lastHeaderCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For col = 1 To lastHeaderCol
        If Not IsError(ws.Cells(1, col).Value2) Then
            If StrComp(CStr(ws.Cells(1, col).Value2), headerText, vbBinaryCompare) = 0 Then
                foundCount = foundCount + 1
                foundCol = col
            End If
        End If
    Next col
    If foundCount > 1 Then
        If createIfMissing Then Err.Raise vbObjectError + 6143, , "同期候補確認履歴の見出しが重複しています: " & headerText
        ReviewAuditColumn = -1
        Exit Function
    End If
    If foundCol = 0 And createIfMissing Then
        Set lastCell = ws.Cells.Find(What:="*", After:=ws.Cells(1, 1), LookIn:=xlFormulas, _
            LookAt:=xlPart, SearchOrder:=xlByColumns, SearchDirection:=xlPrevious, SearchFormat:=False)
        If lastCell Is Nothing Then
            foundCol = 1
        Else
            foundCol = lastCell.Column + 1
        End If
        ws.Cells(1, foundCol).Value2 = headerText
        ws.Cells(1, foundCol).Font.Bold = True
    End If
    ReviewAuditColumn = foundCol
End Function

Private Function ReviewAuditScopeAt(ByVal ws As Worksheet, ByVal rowNo As Long, ByVal scopeCol As Long) As String
    Dim storedScope As String
    If scopeCol < 0 Then
        ReviewAuditScopeAt = "不明"
    ElseIf scopeCol = 0 Then
        ReviewAuditScopeAt = "案件単位"
    ElseIf IsError(ws.Cells(rowNo, scopeCol).Value2) Then
        ReviewAuditScopeAt = "不明"
    Else
        storedScope = Trim$(CStr(ws.Cells(rowNo, scopeCol).Value2))
        If Len(storedScope) = 0 Then storedScope = "案件単位"
        ReviewAuditScopeAt = storedScope
    End If
End Function

Private Function ReviewItemContainsRow(ByVal item As Object, ByVal rowNo As Long) As Boolean
    Dim parts As Variant, part As Variant
    If item Is Nothing Or rowNo < 2 Then Exit Function
    If Not item.Exists("rows") Then Exit Function
    If Len(CStr(item("rows"))) = 0 Then Exit Function
    parts = Split(CStr(item("rows")), ";")
    For Each part In parts
        If CLng(part) = rowNo Then ReviewItemContainsRow = True: Exit Function
    Next part
End Function

Private Function ReviewAuditHashMatches(ByVal stored As Variant, ByVal expected As String) As Boolean
    If IsError(stored) Then Exit Function
    If IsNull(stored) Then Exit Function
    If CStr(stored) = expected Then ReviewAuditHashMatches = True: Exit Function
    If expected = ReviewDigest("") Then
        If IsEmpty(stored) Then ReviewAuditHashMatches = True: Exit Function
        If Len(Trim$(CStr(stored))) = 0 Then ReviewAuditHashMatches = True: Exit Function
        If IsNumeric(stored) Then
            ReviewAuditHashMatches = (CDbl(stored) = 0# Or _
                Abs(CDbl(stored) - CDbl(TimeValue(expected))) < 0.000000001#)
        End If
    End If
End Function

Public Function SyncReviewDecision(ByVal item As Object, Optional ByVal selectedRow As Long = 0) As String
    Dim ws As Worksheet, dst As Worksheet, r As Long, sourceHash As String, destHash As String
    Dim selectedHash As String, storedDecision As String, category As String
    Dim scopeCol As Long, selectedSignatureCol As Long, keyVersionCol As Long
    Dim managementIdCol As Long, stableKeyCol As Long, managementId As String, stableKey As String
    Dim storedManagementId As String, storedStableKey As String, keyVersion As String
    Dim parts As Variant, part As Variant
    Dim allRowsDecided As Boolean, rowCount As Long
    If item Is Nothing Then Exit Function
    category = CStr(item("category"))
    If category = "E" Then
        SyncReviewDecision = CStr(item("decision"))
        If SyncReviewDecision = "未解決" Then SyncReviewDecision = ""
        Exit Function
    End If
    Set ws = ReviewAudit(False)
    If ws Is Nothing Then Exit Function
    sourceHash = ReviewDigest(CStr(item("sourceSignature")))
    If category = "B" And selectedRow >= 2 Then
        If Not ReviewItemContainsRow(item, selectedRow) Then Exit Function
        scopeCol = ReviewAuditColumn(ws, "判断単位")
        selectedSignatureCol = ReviewAuditColumn(ws, "選択Excel行署名")
        keyVersionCol = ReviewAuditColumn(ws, "判断キー形式")
        managementIdCol = ReviewAuditColumn(ws, "選択Excel管理行ID")
        stableKeyCol = ReviewAuditColumn(ws, "CSV安定識別キー")
        If scopeCol <= 0 Or selectedSignatureCol <= 0 Or keyVersionCol < 0 Or _
            managementIdCol < 0 Or stableKeyCol < 0 Then Exit Function
        If Not item.Exists("sourceStableKey") Then Exit Function
        stableKey = CStr(item("sourceStableKey"))
        If Len(stableKey) = 0 Then Exit Function
        Set dst = ThisWorkbook.Worksheets(DEST_SHEET)
        managementId = ReviewManagementRowId(dst, selectedRow)
        If Len(managementId) = 0 Then Exit Function
        selectedHash = ReviewDigest(ReviewSignature(dst, selectedRow))
        For r = ReviewLast(ws) To 2 Step -1
            If ReviewAuditScopeAt(ws, r, scopeCol) = "候補行" And _
               ReviewAuditTextAt(ws, r, 2) = CStr(item("year")) And _
               ReviewAuditTextAt(ws, r, 3) = "B" Then
                keyVersion = "": storedManagementId = "": storedStableKey = ""
                If keyVersionCol > 0 Then
                    If IsError(ws.Cells(r, keyVersionCol).Value2) Then GoTo NextRowAudit
                    If IsNull(ws.Cells(r, keyVersionCol).Value2) Then GoTo NextRowAudit
                    keyVersion = Trim$(ReviewAuditTextAt(ws, r, keyVersionCol))
                End If
                If managementIdCol > 0 Then
                    If IsError(ws.Cells(r, managementIdCol).Value2) Then GoTo NextRowAudit
                    If IsNull(ws.Cells(r, managementIdCol).Value2) Then GoTo NextRowAudit
                    storedManagementId = Trim$(ReviewAuditTextAt(ws, r, managementIdCol))
                End If
                If stableKeyCol > 0 Then
                    If IsError(ws.Cells(r, stableKeyCol).Value2) Then GoTo NextRowAudit
                    If IsNull(ws.Cells(r, stableKeyCol).Value2) Then GoTo NextRowAudit
                    storedStableKey = ReviewAuditTextAt(ws, r, stableKeyCol)
                End If
                If keyVersion = "IDv1" Then
                    If storedManagementId = managementId And storedStableKey = stableKey Then
                        If ReviewAuditTextAt(ws, r, 5) = "別物" Then SyncReviewDecision = "別物"
                        Exit Function
                    End If
                ElseIf Len(keyVersion) = 0 And Len(storedManagementId) = 0 And Len(storedStableKey) = 0 Then
                    If ReviewAuditTextAt(ws, r, 6) = CStr(item("sourceRow")) And _
                       ReviewAuditTextAt(ws, r, 7) = CStr(selectedRow) And _
                       ReviewAuditHashMatches(ws.Cells(r, 9).Value2, sourceHash) And _
                       ReviewAuditHashMatches(ws.Cells(r, selectedSignatureCol).Value2, selectedHash) Then
                        If ReviewAuditTextAt(ws, r, 5) = "別物" Then SyncReviewDecision = "別物"
                        Exit Function
                    End If
                End If
            End If
NextRowAudit:
        Next r
        Exit Function
    End If
    If selectedRow <> 0 Then Exit Function
    destHash = ReviewDigest(CStr(item("destinationSignature")))
    scopeCol = ReviewAuditColumn(ws, "判断単位")
    For r = ReviewLast(ws) To 2 Step -1
        If ReviewAuditScopeAt(ws, r, scopeCol) = "案件単位" And _
           ReviewAuditTextAt(ws, r, 4) = CStr(item("key")) And _
           ReviewAuditHashMatches(ws.Cells(r, 9).Value2, sourceHash) And _
           ReviewAuditHashMatches(ws.Cells(r, 10).Value2, destHash) Then
            storedDecision = ReviewAuditTextAt(ws, r, 5)
            Select Case storedDecision
                Case "別物", "保留", "同一として処理", "削除完了"
                    If category = "B" And storedDecision = "別物" Then Exit For
                    SyncReviewDecision = storedDecision
                    Exit Function
            End Select
        End If
    Next r
    If category <> "B" Or Len(CStr(item("rows"))) = 0 Then Exit Function
    parts = Split(CStr(item("rows")), ";")
    rowCount = UBound(parts) - LBound(parts) + 1
    allRowsDecided = (rowCount > 0)
    For Each part In parts
        If Len(Trim$(CStr(part))) = 0 Then
            allRowsDecided = False
            Exit For
        End If
        If SyncReviewDecision(item, CLng(part)) <> "別物" Then
            allRowsDecided = False
            Exit For
        End If
    Next part
    If allRowsDecided Then SyncReviewDecision = "候補行確認済"
End Function

Public Function SyncReviewDecisionComment(ByVal item As Object, Optional ByVal selectedRow As Long = 0) As String
    Dim ws As Worksheet, dst As Worksheet, r As Long, sourceHash As String, destHash As String
    Dim selectedHash As String, storedDecision As String, scopeCol As Long, selectedSignatureCol As Long
    Dim keyVersionCol As Long, managementIdCol As Long, stableKeyCol As Long
    Dim managementId As String, stableKey As String, storedManagementId As String
    Dim storedStableKey As String, keyVersion As String
    If item Is Nothing Then Exit Function
    Set ws = ReviewAudit(False)
    If ws Is Nothing Then Exit Function
    sourceHash = ReviewDigest(CStr(item("sourceSignature")))
    If CStr(item("category")) = "B" And selectedRow >= 2 Then
        If Not ReviewItemContainsRow(item, selectedRow) Then Exit Function
        scopeCol = ReviewAuditColumn(ws, "判断単位")
        selectedSignatureCol = ReviewAuditColumn(ws, "選択Excel行署名")
        keyVersionCol = ReviewAuditColumn(ws, "判断キー形式")
        managementIdCol = ReviewAuditColumn(ws, "選択Excel管理行ID")
        stableKeyCol = ReviewAuditColumn(ws, "CSV安定識別キー")
        If scopeCol <= 0 Or selectedSignatureCol <= 0 Or keyVersionCol < 0 Or _
            managementIdCol < 0 Or stableKeyCol < 0 Then Exit Function
        If Not item.Exists("sourceStableKey") Then Exit Function
        stableKey = CStr(item("sourceStableKey"))
        If Len(stableKey) = 0 Then Exit Function
        Set dst = ThisWorkbook.Worksheets(DEST_SHEET)
        managementId = ReviewManagementRowId(dst, selectedRow)
        If Len(managementId) = 0 Then Exit Function
        selectedHash = ReviewDigest(ReviewSignature(dst, selectedRow))
        For r = ReviewLast(ws) To 2 Step -1
            If ReviewAuditScopeAt(ws, r, scopeCol) = "候補行" And _
               ReviewAuditTextAt(ws, r, 2) = CStr(item("year")) And _
               ReviewAuditTextAt(ws, r, 3) = "B" Then
                keyVersion = "": storedManagementId = "": storedStableKey = ""
                If keyVersionCol > 0 Then
                    If IsError(ws.Cells(r, keyVersionCol).Value2) Then GoTo NextCommentAudit
                    If IsNull(ws.Cells(r, keyVersionCol).Value2) Then GoTo NextCommentAudit
                    keyVersion = Trim$(ReviewAuditTextAt(ws, r, keyVersionCol))
                End If
                If managementIdCol > 0 Then
                    If IsError(ws.Cells(r, managementIdCol).Value2) Then GoTo NextCommentAudit
                    If IsNull(ws.Cells(r, managementIdCol).Value2) Then GoTo NextCommentAudit
                    storedManagementId = Trim$(ReviewAuditTextAt(ws, r, managementIdCol))
                End If
                If stableKeyCol > 0 Then
                    If IsError(ws.Cells(r, stableKeyCol).Value2) Then GoTo NextCommentAudit
                    If IsNull(ws.Cells(r, stableKeyCol).Value2) Then GoTo NextCommentAudit
                    storedStableKey = ReviewAuditTextAt(ws, r, stableKeyCol)
                End If
                If keyVersion = "IDv1" Then
                    If storedManagementId = managementId And storedStableKey = stableKey Then
                        If ReviewAuditTextAt(ws, r, 5) = "別物" Then _
                            SyncReviewDecisionComment = ReviewAuditTextAt(ws, r, 11)
                        Exit Function
                    End If
                ElseIf Len(keyVersion) = 0 And Len(storedManagementId) = 0 And Len(storedStableKey) = 0 Then
                    If ReviewAuditTextAt(ws, r, 6) = CStr(item("sourceRow")) And _
                       ReviewAuditTextAt(ws, r, 7) = CStr(selectedRow) And _
                       ReviewAuditHashMatches(ws.Cells(r, 9).Value2, sourceHash) And _
                       ReviewAuditHashMatches(ws.Cells(r, selectedSignatureCol).Value2, selectedHash) Then
                        If ReviewAuditTextAt(ws, r, 5) = "別物" Then _
                            SyncReviewDecisionComment = ReviewAuditTextAt(ws, r, 11)
                        Exit Function
                    End If
                End If
            End If
NextCommentAudit:
        Next r
        Exit Function
    End If
    If selectedRow <> 0 Then Exit Function
    destHash = ReviewDigest(CStr(item("destinationSignature")))
    scopeCol = ReviewAuditColumn(ws, "判断単位")
    For r = ReviewLast(ws) To 2 Step -1
        If ReviewAuditScopeAt(ws, r, scopeCol) = "案件単位" And _
           ReviewAuditTextAt(ws, r, 4) = CStr(item("key")) And _
           ReviewAuditHashMatches(ws.Cells(r, 9).Value2, sourceHash) And _
           ReviewAuditHashMatches(ws.Cells(r, 10).Value2, destHash) Then
            storedDecision = ReviewAuditTextAt(ws, r, 5)
            Select Case storedDecision
                Case "別物", "保留", "同一として処理", "削除完了"
                    SyncReviewDecisionComment = ReviewAuditTextAt(ws, r, 11)
                    Exit Function
            End Select
        End If
    Next r
End Function

Public Function SyncReviewLegacyCaseDecision(ByVal item As Object) As String
    Dim ws As Worksheet, r As Long, sourceHash As String, destHash As String
    Dim scopeCol As Long, storedDecision As String
    If item Is Nothing Then Exit Function
    If CStr(item("category")) <> "B" Then Exit Function
    Set ws = ReviewAudit(False)
    If ws Is Nothing Then Exit Function
    sourceHash = ReviewDigest(CStr(item("sourceSignature")))
    destHash = ReviewDigest(CStr(item("destinationSignature")))
    scopeCol = ReviewAuditColumn(ws, "判断単位")
    For r = ReviewLast(ws) To 2 Step -1
        If ReviewAuditScopeAt(ws, r, scopeCol) = "案件単位" And _
           ReviewAuditTextAt(ws, r, 4) = CStr(item("key")) And _
           ReviewAuditHashMatches(ws.Cells(r, 9).Value2, sourceHash) And _
           ReviewAuditHashMatches(ws.Cells(r, 10).Value2, destHash) Then
            storedDecision = ReviewAuditTextAt(ws, r, 5)
            Select Case storedDecision
                Case "別物"
                    SyncReviewLegacyCaseDecision = "別物"
                    Exit Function
                Case "保留", "同一として処理", "削除完了"
                    Exit Function
            End Select
        End If
    Next r
End Function

Private Sub ReviewWriteAudit(ByVal item As Object, ByVal decision As String, ByVal detail As String, _
                             ByVal backup As String, Optional ByVal selectedRow As Long = 0)
    Dim ws As Worksheet, dst As Worksheet, r As Long, scopeCol As Long, selectedSignatureCol As Long
    Dim keyVersionCol As Long, managementIdCol As Long, stableKeyCol As Long
    Dim selectedHash As String, managementId As String, stableKey As String
    If selectedRow > 0 Then
        If CStr(item("category")) <> "B" Or decision <> "別物" Then _
            Err.Raise vbObjectError + 6144, , "候補行単位で記録できるのはB候補の「別物」だけです。"
        If Not ReviewItemContainsRow(item, selectedRow) Then _
            Err.Raise vbObjectError + 6145, , "選択したExcel行は現在の候補に含まれていません。"
        Set dst = ThisWorkbook.Worksheets(DEST_SHEET)
        managementId = ReviewManagementRowId(dst, selectedRow)
        If Len(managementId) = 0 Then _
            Err.Raise vbObjectError + 6149, , "選択行のExcel管理行IDが未付与・不正・重複しています。IDを点検してから判断を記録してください。"
        If Not item.Exists("sourceStableKey") Then _
            Err.Raise vbObjectError + 6150, , "CSV側の正式識別情報を安全に一意特定できません。"
        stableKey = CStr(item("sourceStableKey"))
        If Len(stableKey) = 0 Then _
            Err.Raise vbObjectError + 6150, , "CSV側の正式識別情報が欠損・重複しています。判断を記録できません。"
        selectedHash = ReviewDigest(ReviewSignature(dst, selectedRow))
    End If
    Set ws = ReviewAudit(True, selectedRow > 0)
    scopeCol = ReviewAuditColumn(ws, "判断単位")
    selectedSignatureCol = ReviewAuditColumn(ws, "選択Excel行署名")
    keyVersionCol = ReviewAuditColumn(ws, "判断キー形式")
    managementIdCol = ReviewAuditColumn(ws, "選択Excel管理行ID")
    stableKeyCol = ReviewAuditColumn(ws, "CSV安定識別キー")
    If selectedRow > 0 And (scopeCol <= 0 Or selectedSignatureCol <= 0 Or _
        keyVersionCol <= 0 Or managementIdCol <= 0 Or stableKeyCol <= 0) Then _
        Err.Raise vbObjectError + 6146, , "同期候補確認履歴の行別判断列を確認できません。"
    r = ReviewLast(ws) + 1
    If selectedRow > 0 Then
        ws.Cells(r, keyVersionCol).Value2 = "IDv1"
        ws.Cells(r, scopeCol).Value2 = "候補行"
        ws.Cells(r, managementIdCol).NumberFormat = "@"
        ws.Cells(r, managementIdCol).Value2 = managementId
        ws.Cells(r, stableKeyCol).NumberFormat = "@"
        ws.Cells(r, stableKeyCol).Value2 = stableKey
    End If
    ws.Cells(r, 1).Value2 = Now
    ws.Cells(r, 2).Value2 = CLng(item("year"))
    ws.Cells(r, 3).Value2 = CStr(item("category"))
    ws.Cells(r, 4).NumberFormat = "@"
    ws.Cells(r, 9).NumberFormat = "@": ws.Cells(r, 10).NumberFormat = "@"
    ws.Cells(r, 4).Value2 = CStr(item("key"))
    ws.Cells(r, 5).Value2 = decision
    ws.Cells(r, 6).Value2 = CLng(item("sourceRow"))
    If selectedRow > 0 Then
        ws.Cells(r, 7).NumberFormat = "@"
        ws.Cells(r, 7).Value2 = CStr(selectedRow)
    Else
        ws.Cells(r, 7).Value2 = CStr(item("rows"))
    End If
    ws.Cells(r, 8).Value2 = CStr(item("reason"))
    ws.Cells(r, 9).Value2 = ReviewDigest(CStr(item("sourceSignature")))
    ws.Cells(r, 10).Value2 = ReviewDigest(CStr(item("destinationSignature")))
    ws.Cells(r, 11).Value2 = detail
    ws.Cells(r, 12).Value2 = backup
    If selectedRow > 0 Then
        ws.Cells(r, selectedSignatureCol).NumberFormat = "@"
        ws.Cells(r, selectedSignatureCol).Value2 = selectedHash
    Else
        If scopeCol > 0 Then ws.Cells(r, scopeCol).Value2 = "案件単位"
        If selectedSignatureCol > 0 Then ws.Cells(r, selectedSignatureCol).Value2 = ""
        If keyVersionCol > 0 Then ws.Cells(r, keyVersionCol).Value2 = ""
        If managementIdCol > 0 Then ws.Cells(r, managementIdCol).Value2 = ""
        If stableKeyCol > 0 Then ws.Cells(r, stableKeyCol).Value2 = ""
    End If
End Sub
Private Function ReviewCurrent(ByVal item As Object, Optional ByVal current As Collection) As Boolean
    Dim candidate As Object, i As Long
    If current Is Nothing Then Set current = SyncReviewCandidates(CLng(item("year")))
    For i = 1 To current.Count
        Set candidate = current(i)
        If CStr(candidate("key")) = CStr(item("key")) And _
           CStr(candidate("sourceSignature")) = CStr(item("sourceSignature")) And _
           CStr(candidate("destinationSignature")) = CStr(item("destinationSignature")) And _
           CStr(candidate("destinationIdSnapshot")) = CStr(item("destinationIdSnapshot")) And _
           CStr(candidate("sourceStableKey")) = CStr(item("sourceStableKey")) Then
            ReviewCurrent = True
            Exit Function
        End If
    Next i
End Function

Public Sub SyncReviewNavigate(ByVal item As Object, ByVal selectedRow As Long)
    Dim dst As Worksheet
    Set dst = ThisWorkbook.Worksheets(DEST_SHEET)
    If selectedRow < 2 Or selectedRow > ReviewLast(dst) Or Not ReviewCurrent(item) Then _
        Err.Raise vbObjectError + 6106, , "対象行が移動・変更されました。候補を再読み込みしてください。"
    If InStr(1, ";" & CStr(item("rows")) & ";", ";" & CStr(selectedRow) & ";", vbBinaryCompare) = 0 Then _
        Err.Raise vbObjectError + 6107, , "候補外の行へは移動できません。"
    Application.Goto dst.Cells(selectedRow, 1), True
End Sub

Public Function SyncReviewDecide(ByVal item As Object, ByVal decision As String, _
                                 Optional ByVal selectedRow As Long = 0) As Boolean
    On Error GoTo Failed
    If decision <> "別物" And decision <> "保留" Then Err.Raise vbObjectError + 6108, , "記録できない判断です。"
    If CStr(item("category")) = "B" And decision = "別物" Then
        If selectedRow < 2 Or Not ReviewItemContainsRow(item, selectedRow) Then _
            Err.Raise vbObjectError + 6147, , "B候補の「別物」は選択したExcel候補行を指定してください。"
    ElseIf selectedRow <> 0 Then
        Err.Raise vbObjectError + 6148, , "選択行単位で記録できるのはB候補の「別物」だけです。"
    End If
    If Not ReviewCurrent(item) Then Err.Raise vbObjectError + 6109, , "CSVまたはExcelの行・内容・候補が変わりました。再読み込みしてください。"
    If selectedRow > 0 Then
        ReviewWriteAudit item, decision, "データ変更なし。選択したExcel行だけを別物として記録。次回の変更時には再判定", "", selectedRow
    Else
        ReviewWriteAudit item, decision, "データ変更なし。次回の変更時には再判定", ""
    End If
    SyncReviewDecide = True
    Exit Function
Failed:
    MsgBox "判断を記録できません: " & Err.Description, vbExclamation
End Function

Public Function SyncReviewApprove(ByVal item As Object, ByVal selectedRow As Long) As Boolean
    Dim src As Worksheet, dst As Worksheet, sRow As Long, id As String, no As String, bar As String
    Dim oldText As String, newText As String, backup As String, failureText As String
    Dim i As Long, prevEvents As Boolean, changing As Boolean
    Dim cols(1 To 4) As Long, oldValues(1 To 4) As Variant, oldFormats(1 To 4) As String, j As Long
    Dim oldFormulas(1 To 4) As String, wasFormula(1 To 4) As Boolean
    On Error GoTo Failed
    If CStr(item("category")) <> "B" Then Err.Raise vbObjectError + 6110, , "ID競合や複数行の統合は自動承認できません。手動で確認してください。"
    If Not ReviewCurrent(item) Then Err.Raise vbObjectError + 6109, , "候補・行・データが変わりました。再読み込みしてください。"
    Set src = ThisWorkbook.Worksheets(CSV_SHEET): Set dst = ThisWorkbook.Worksheets(DEST_SHEET)
    sRow = CLng(item("sourceRow"))
    If CStr(item("rows")) <> CStr(selectedRow) Then Err.Raise vbObjectError + 6111, , "候補が複数あります。既存行の統合は行いません。"
    If ReviewFiscal(src, sRow) <> CLng(item("year")) Or ReviewFiscal(dst, selectedRow) <> CLng(item("year")) Then _
        Err.Raise vbObjectError + 6112, , "年度が一致しません。"
    If ReviewField(dst, selectedRow, "同期状態") <> "登録予定" Or Len(ReviewField(dst, selectedRow, "ファイルID")) > 0 Then _
        Err.Raise vbObjectError + 6113, , "登録予定かつID空欄の一意な行に限ります。"
    If Len(ReviewName(src, sRow, True, False)) = 0 Or ReviewName(src, sRow, True, False) <> ReviewName(dst, selectedRow, False, False) Then _
        Err.Raise vbObjectError + 6114, , "作成年度・ガイドを一意に照合できません。"
    id = ReviewField(src, sRow, "ファイルID")
    no = ReviewField(src, sRow, "ファイル番号")
    bar = ReviewField(src, sRow, "バーコード番号")
    If Len(id) = 0 Or Len(no) = 0 Or Len(bar) = 0 Then Err.Raise vbObjectError + 6115, , "3つの識別番号が揃っていません。"
    For i = 2 To ReviewLast(src)
        If i <> sRow Then
            If ReviewField(src, i, "ファイルID") = id Or ReviewField(src, i, "バーコード番号") = bar Or _
               ReviewPadding(ReviewField(src, i, "ファイル番号")) = ReviewPadding(no) Then _
               Err.Raise vbObjectError + 6116, , "CSV側の識別番号が一意ではありません。"
        End If
    Next i
    For i = 2 To ReviewLast(dst)
        If i <> selectedRow Then
            If ReviewField(dst, i, "ファイルID") = id Or ReviewField(dst, i, "バーコード番号") = bar Or _
               ReviewPadding(ReviewField(dst, i, "ファイル番号")) = ReviewPadding(no) Then _
               Err.Raise vbObjectError + 6117, , "他のExcel行と識別番号が重なります。統合せず保留してください。"
        End If
    Next i
    If Len(ReviewField(dst, selectedRow, "ファイル番号")) > 0 And _
       ReviewPadding(ReviewField(dst, selectedRow, "ファイル番号")) <> ReviewPadding(no) Then _
       Err.Raise vbObjectError + 6118, , "登録予定行のファイル番号が異なります。"
    If Len(ReviewField(dst, selectedRow, "バーコード番号")) > 0 And _
       ReviewField(dst, selectedRow, "バーコード番号") <> bar Then _
       Err.Raise vbObjectError + 6119, , "登録予定行のバーコードが異なります。"
    oldText = ReviewDescription(dst, selectedRow, False)
    newText = "Excel行" & selectedRow & " の正式ID [空欄]→" & id & _
        "、ファイル番号 [" & ReviewField(dst, selectedRow, "ファイル番号") & "]→" & no & _
        "、バーコード [" & ReviewField(dst, selectedRow, "バーコード番号") & "]→" & bar & _
        "、同期状態 [登録予定]→[同期済]。ガイド・フォルダー・Excel独自列は変更しません。"
    If MsgBox(oldText & vbCrLf & ReviewDescription(src, sRow, True) & vbCrLf & vbCrLf & newText & vbCrLf & _
        "実行しますか？", vbYesNo + vbQuestion + vbDefaultButton2, "最終確認") <> vbYes Then Exit Function
    If Not ReviewCurrent(item) Then Err.Raise vbObjectError + 6120, , "確認中にCSVまたはExcelの内容が変わりました。"
    If ThisWorkbook.ReadOnly Or dst.ProtectContents Then Err.Raise vbObjectError + 6121, , "ブックが読み取り専用またはシートが保護されています。"
    cols(1) = ReviewCol(dst, "ファイルID"): cols(2) = ReviewCol(dst, "ファイル番号")
    cols(3) = ReviewCol(dst, "バーコード番号"): cols(4) = ReviewCol(dst, "同期状態")
    For j = 1 To 4
        oldValues(j) = dst.Cells(selectedRow, cols(j)).Value2
        wasFormula(j) = dst.Cells(selectedRow, cols(j)).HasFormula
        If wasFormula(j) Then oldFormulas(j) = dst.Cells(selectedRow, cols(j)).Formula
        oldFormats(j) = dst.Cells(selectedRow, cols(j)).NumberFormat
    Next j
    backup = ReviewBackup()
    If Not ReviewCurrent(item) Then Err.Raise vbObjectError + 6137, , "バックアップ後に候補が変わりました。"
    新ファイル基準表_ファイルID編集スナップショットを無効化
    prevEvents = Application.EnableEvents: Application.EnableEvents = False: changing = True
    For j = 1 To 3: dst.Cells(selectedRow, cols(j)).NumberFormat = "@": Next j
    dst.Cells(selectedRow, cols(1)).Value2 = id
    dst.Cells(selectedRow, cols(2)).Value2 = no
    dst.Cells(selectedRow, cols(3)).Value2 = bar
    dst.Cells(selectedRow, cols(4)).Value2 = "同期済"
    ReviewWriteAudit item, "同一として処理", newText, backup
    changing = False: Application.EnableEvents = prevEvents
    SyncReviewApprove = True
    Exit Function
Failed:
    failureText = Err.Description
    If changing Then
        On Error Resume Next
        For j = 1 To 4
            dst.Cells(selectedRow, cols(j)).NumberFormat = oldFormats(j)
            If wasFormula(j) Then
                dst.Cells(selectedRow, cols(j)).Formula = oldFormulas(j)
            Else
                dst.Cells(selectedRow, cols(j)).Value2 = oldValues(j)
            End If
        Next j
        Application.EnableEvents = prevEvents
        On Error GoTo 0
    End If
    MsgBox "承認を実行できません: " & failureText & vbCrLf & "バックアップ: " & backup & vbCrLf & _
        "編集済みなら保存前に対象行と履歴を確認してください。", vbCritical
End Function

Private Function ReviewBackup(Optional ByVal backupPurpose As String = "削除前") As String
    Dim original As String, base As String, ext As String, target As String, suffix As Long, dotPos As Long
    original = ThisWorkbook.FullName
    If Len(ThisWorkbook.Path) = 0 Or ThisWorkbook.ReadOnly Then _
        Err.Raise vbObjectError + 6122, , "保存済みの編集可能なブックで実行してください。"
    dotPos = InStrRev(original, ".")
    If dotPos = 0 Then Err.Raise vbObjectError + 6123, , "バックアップの拡張子を判断できません。"
    base = Left$(original, dotPos - 1): ext = Mid$(original, dotPos)
    target = base & "_同期候補" & backupPurpose & "_" & Format$(Now, "yyyymmdd_hhnnss") & ext
    Do While Len(Dir$(target)) > 0
        suffix = suffix + 1
        target = base & "_同期候補" & backupPurpose & "_" & Format$(Now, "yyyymmdd_hhnnss") & "_" & suffix & ext
    Loop
    ThisWorkbook.SaveCopyAs target
    If Len(Dir$(target)) = 0 Then Err.Raise vbObjectError + 6124, , "バックアップを作成できませんでした。"
    ReviewBackup = target
End Function

Public Function SyncReviewDeleteBatch(ByVal selected As Collection) As Boolean
    Dim dst As Worksheet, src As Worksheet, item As Object, ordered As New Collection
    Dim i As Long, j As Long, bestRow As Long, bestIndex As Long, rowNo As Long
    Dim selectedKeys As Object, summary As String, backup As String, prevEvents As Boolean, changing As Boolean
    Dim completed As Long, failureText As String, current As Collection, approvedKeys As Object
    Dim rowDeleted As Boolean
    On Error GoTo Failed
    If selected Is Nothing Then Err.Raise vbObjectError + 6125, , "削除候補を選択してください。"
    If selected.Count = 0 Then Err.Raise vbObjectError + 6125, , "削除候補を選択してください。"
    Set dst = ThisWorkbook.Worksheets(DEST_SHEET): Set src = ThisWorkbook.Worksheets(CSV_SHEET)
    If ThisWorkbook.ReadOnly Or dst.ProtectContents Or dst.FilterMode Then _
        Err.Raise vbObjectError + 6126, , "読み取り専用・シート保護・フィルター適用中は削除しません。"
    Set selectedKeys = CreateObject("Scripting.Dictionary")
    Set current = SyncReviewCandidates(CLng(selected(1)("year")))
    For Each item In selected
        If CStr(item("category")) <> "C" Or CLng(item("sourceRow")) <> 0 Then _
            Err.Raise vbObjectError + 6127, , "削除可以外の候補が混ざっています。"
        rowNo = CLng(item("rows"))
        If selectedKeys.Exists(CStr(rowNo)) Then Err.Raise vbObjectError + 6128, , "同じ行を二重に選択しています。"
        selectedKeys.Add CStr(rowNo), True
        If Len(SyncReviewDecision(item)) > 0 Then Err.Raise vbObjectError + 6138, , "保留・別物と判断済みの候補は削除できません。"
        If Not ReviewCurrent(item, current) Then Err.Raise vbObjectError + 6129, , "CSV・対象行・候補が変更されました。再読み込みしてください。"
        summary = summary & vbCrLf & ReviewDescription(dst, rowNo, False)
    Next item
    If MsgBox("次の " & selected.Count & " 行を完全に削除します。取り消しは事前バックアップから復旧してください。" & _
        vbCrLf & summary & vbCrLf & vbCrLf & "本当に一括削除しますか？", _
        vbYesNo + vbExclamation + vbDefaultButton2, "最終確認：一括削除") <> vbYes Then Exit Function
    Set current = SyncReviewCandidates(CLng(selected(1)("year")))
    For Each item In selected
        If Len(SyncReviewDecision(item)) > 0 Or Not ReviewCurrent(item, current) Then _
            Err.Raise vbObjectError + 6130, , "最終確認中に判断または内容が変わりました。"
    Next item
    backup = ReviewBackup()
    Set approvedKeys = CreateObject("Scripting.Dictionary")
    For Each item In selected
        ReviewWriteAudit item, "削除承認", "最終確認で選択・同意済み。データは未変更。", backup
        approvedKeys.Add CStr(item("key")), True
    Next item
    For i = selected.Count To 1 Step -1
        bestRow = 0: bestIndex = 0
        For j = 1 To selected.Count
            Set item = selected(j)
            rowNo = CLng(item("rows"))
            If Not ReviewCollectionHasRow(ordered, rowNo) And rowNo > bestRow Then
                bestRow = rowNo: bestIndex = j
            End If
        Next j
        If bestIndex > 0 Then ordered.Add selected(bestIndex)
    Next i
    prevEvents = Application.EnableEvents: Application.EnableEvents = False: changing = True
    For Each item In ordered
        rowDeleted = False
        rowNo = CLng(item("rows"))
        If Not approvedKeys.Exists(CStr(item("key"))) Or Len(SyncReviewDecision(item)) > 0 Then _
            Err.Raise vbObjectError + 6131, , "未承認または保留済みの候補です。残りは停止します。"
        If ReviewSignature(dst, rowNo) <> CStr(item("destinationSignature")) Then _
            Err.Raise vbObjectError + 6131, , "対象行の内容が変わりました。残りは停止します。"
        ReviewWriteAudit item, "削除着手", ReviewDescription(dst, rowNo, False), backup
        dst.Rows(rowNo).Delete
        rowDeleted = True
        completed = completed + 1
        ReviewWriteAudit item, "削除完了", "Excel行" & rowNo & "を削除。復元はバックアップから", backup
        rowDeleted = False
    Next item
    changing = False: Application.EnableEvents = prevEvents
    MsgBox completed & " 行を削除しました。履歴シートと事前バックアップを確認して保存してください。" & vbCrLf & backup, vbInformation
    SyncReviewDeleteBatch = True
    Exit Function
Failed:
    failureText = Err.Description
    If changing Then Application.EnableEvents = prevEvents
    If Len(backup) > 0 Then
        On Error Resume Next
        Err.Clear
        If rowDeleted Then
            ReviewWriteAudit item, "削除完了", "Excel行" & rowNo & "は削除済み。最初の完了履歴追記に失敗: " & failureText, backup
        Else
            ReviewWriteAudit item, "削除失敗", failureText & " / 実削除件数 " & completed, backup
        End If
        If Err.Number <> 0 Then failureText = failureText & " / 監査履歴への再記録にも失敗: " & Err.Description
        On Error GoTo 0
    End If
    MsgBox "一括削除を停止しました（物理削除済み " & completed & " 件）。" & failureText & vbCrLf & _
        "バックアップ: " & backup & vbCrLf & "続行せず、履歴と対象行を確認してください。", vbCritical
End Function

Private Function ReviewCollectionHasRow(ByVal rows As Collection, ByVal rowNo As Long) As Boolean
    Dim item As Object
    For Each item In rows
        If CLng(item("rows")) = rowNo Then ReviewCollectionHasRow = True: Exit Function
    Next item
End Function
