Option Explicit

Private mSelectingRowFromMoveMode As Boolean
Private mRowIdSelectionSnapshot As Object
Private mFileIdSelectionSnapshot As Object
Private mFileIdSelectionSnapshotEpoch As Double

'========================================================
' 貼り付け先:
'   VBAProject
'   └ Microsoft Excel Objects
'      └ 新ファイル基準表
'
' ※標準モジュールではなく、必ず「新ファイル基準表」の
'   シートコードモジュールへ配置してください。
'========================================================
Private Sub Worksheet_Activate()

    On Error Resume Next
    Set mRowIdSelectionSnapshot = ExcelRowIdCaptureSelection(Application.ActiveCell, Me)
    CaptureFileIdSelectionSnapshot Application.ActiveCell, Me
    On Error GoTo 0

End Sub

Private Sub Worksheet_Change(ByVal Target As Range)

    Dim oldEnableEvents As Boolean

    oldEnableEvents = Application.EnableEvents
    On Error GoTo ExitHandler
    If Target Is Nothing Then Exit Sub
    If Intersect(Target, Me.Rows(1)) Is Nothing Then
        ' OK
    ElseIf Target.Rows.Count = 1 And Target.Row = 1 Then
        Exit Sub
    End If

    Application.EnableEvents = False

    ' U列「ファイルID」は管理システムの正式識別値のため、利用者の直接編集・貼り付けを復元する。
    ' CSV取込・同期マクロは呼出元でイベントを停止して正当な更新を行う。
    If 新ファイル基準表_ファイルID直接編集を復元(Target, Me) Then GoTo ExitHandler

    ' Worksheet_Change が発火した手動編集を記録する。自動処理は呼出元でイベントを停止する。
    新ファイル基準表_最終編集情報を記録 Target, Me

    ' 標準モジュール(Module1)の補完処理を呼び出す
    ' ※この中で「年度（和暦）→年度（西暦）」および
    '   作成年度の西暦変換、および廃棄年月日・予定年の連動まで実行する
    新ファイル基準表_表示名からID自動補完 Target
    ' 登録予定行では、コピー元の識別番号や直接入力された番号を残さない。
    新ファイル基準表_登録予定識別番号を自動クリア Target, Me
    ' 追記: 色名が変更された行は 2色名 へ同値同期
    ' 新ファイル基準表_色名から2色名同期 Target, Me

    ' 呼び出し先でイベント状態が変わる可能性に備えて再度OFF
    Application.EnableEvents = False

    ' タイトル/主要編集列の変更時は、通し番号を自動で再採番
    ExcelRowIdHandleWorksheetChange Target, Me, mRowIdSelectionSnapshot

    If ShouldReindexSerial(Target, Me) Then
        ReindexSerial_NewFileStandard Me
    End If

    If TypeName(Application.Selection) = "Range" Then
        Set mRowIdSelectionSnapshot = ExcelRowIdCaptureSelection(Application.Selection, Me)
        CaptureFileIdSelectionSnapshot Application.Selection, Me
    End If

ExitHandler:
    Application.EnableEvents = oldEnableEvents

End Sub

Private Sub Worksheet_SelectionChange(ByVal Target As Range)

    Dim oldEnableEvents As Boolean
    Dim firstRow As Long
    Dim lastRow As Long
    Dim anchorRow As Long
    Dim clickedRow As Long
    Dim assistMode As Boolean
    Dim clickMode As Boolean
    Dim targetRow As Long

    On Error GoTo ExitHandler

    oldEnableEvents = Application.EnableEvents
    If Target Is Nothing Then GoTo ExitHandler
    Set mRowIdSelectionSnapshot = ExcelRowIdCaptureSelection(Target, Me)
    CaptureFileIdSelectionSnapshot Target, Me
    If mSelectingRowFromMoveMode Then GoTo ExitHandler
    clickMode = IsRowMoveClickModeActive()
    If (Not IsRowMoveModeActive()) And (Not clickMode) Then GoTo ExitHandler
    If Target.Areas.Count <> 1 Then GoTo ExitHandler
    If Target.Row < 2 Then GoTo ExitHandler

    If clickMode Then
        targetRow = Target.Row
        mSelectingRowFromMoveMode = True
        Application.EnableEvents = False
        行移動_移動先クリックで確定 targetRow
        GoTo ExitHandler
    End If

    firstRow = Target.Row
    lastRow = Target.Row + Target.Rows.Count - 1
    If lastRow < firstRow Then GoTo ExitHandler

    assistMode = IsRowMoveSelectionAssistActive()
    If assistMode Then
        anchorRow = GetRowMoveSelectionAnchorRow()
        clickedRow = Target.Row
        If anchorRow < 2 Then anchorRow = clickedRow

        If clickedRow >= anchorRow Then
            firstRow = anchorRow
            lastRow = clickedRow
        Else
            firstRow = clickedRow
            lastRow = anchorRow
        End If

        UpdateRowMoveSelectionRange firstRow, lastRow
    End If

    If (Not assistMode) And Target.Columns.Count = Me.Columns.Count Then GoTo ExitHandler

    mSelectingRowFromMoveMode = True
    Application.EnableEvents = False
    Me.Rows(firstRow & ":" & lastRow).Select

ExitHandler:
    Application.EnableEvents = oldEnableEvents
    mSelectingRowFromMoveMode = False

End Sub

Private Sub CaptureFileIdSelectionSnapshot(ByVal Target As Range, ByVal ws As Worksheet)

    On Error GoTo ClearSnapshot
    Set mFileIdSelectionSnapshot = NewFileStandardCaptureFileIdSelection(Target, ws)
    mFileIdSelectionSnapshotEpoch = 新ファイル基準表_ファイルID編集スナップショット世代()
    Exit Sub

ClearSnapshot:
    Set mFileIdSelectionSnapshot = Nothing
    mFileIdSelectionSnapshotEpoch = 新ファイル基準表_ファイルID編集スナップショット世代()

End Sub

Private Function 新ファイル基準表_ファイルID直接編集を復元( _
    ByVal Target As Range, ByVal ws As Worksheet) As Boolean

    Dim fileIdCol As Long, syncCol As Long
    Dim changedFileIdRange As Range, changedCell As Range
    Dim rowsToRestore As Object, attemptedSnapshots As Object
    Dim rowKey As Variant, rowNo As Long
    Dim originalSnapshot As Object

    If Target Is Nothing Or ws Is Nothing Then Exit Function
    If Not (Target.Worksheet Is ws) Then Exit Function
    fileIdCol = NewFileStandardFindUniqueHeaderColumn(ws, "ファイルID")
    If fileIdCol <= 0 Then Exit Function
    Set changedFileIdRange = Application.Intersect(Target, ws.Columns(fileIdCol))
    If changedFileIdRange Is Nothing Then Exit Function

    If mFileIdSelectionSnapshot Is Nothing Or _
       mFileIdSelectionSnapshotEpoch <> 新ファイル基準表_ファイルID編集スナップショット世代() Then
        新ファイル基準表_ファイルID直接編集を復元 = UndoUnsafeFileIdEdit(Target, ws)
        Exit Function
    End If

    Set rowsToRestore = CreateObject("Scripting.Dictionary")
    rowsToRestore.CompareMode = vbTextCompare
    Set attemptedSnapshots = CreateObject("Scripting.Dictionary")
    attemptedSnapshots.CompareMode = vbTextCompare

    For Each changedCell In changedFileIdRange.Cells
        If changedCell.Row >= 2 Then
            rowKey = CStr(changedCell.Row)
            If Not mFileIdSelectionSnapshot.Exists(CStr(rowKey)) Then
                新ファイル基準表_ファイルID直接編集を復元 = UndoUnsafeFileIdEdit(Target, ws)
                Exit Function
            End If
            Set originalSnapshot = mFileIdSelectionSnapshot.Item(CStr(rowKey))
            If Not FileIdCellMatchesSnapshot(changedCell, originalSnapshot) Then
                If Not rowsToRestore.Exists(CStr(rowKey)) Then
                    rowsToRestore.Add CStr(rowKey), originalSnapshot
                    attemptedSnapshots.Add CStr(rowKey), CaptureFileIdCellSnapshot(changedCell)
                End If
            End If
        End If
    Next changedCell

    If rowsToRestore.Count = 0 Then Exit Function

    ' 同じ貼り付けで同期状態を「登録予定」に変えると、既存仕様はU/V/Wを消去する。
    ' 既存IDを保持しながらU列だけを安全に戻せないため、操作全体を取り消す。
    syncCol = NewFileStandardFindUniqueHeaderColumn(ws, "同期状態")
    If syncCol > 0 Then
        For Each rowKey In rowsToRestore.Keys
            rowNo = CLng(rowKey)
            If Not (Application.Intersect(Target, ws.Cells(rowNo, syncCol)) Is Nothing) Then
                If Trim$(CStr(ws.Cells(rowNo, syncCol).Value2)) = "登録予定" Then
                    新ファイル基準表_ファイルID直接編集を復元 = UndoUnsafeFileIdEdit(Target, ws)
                    Exit Function
                End If
            End If
        Next rowKey
    End If

    On Error GoTo RestoreFailed
    For Each rowKey In rowsToRestore.Keys
        RestoreFileIdCellFromSnapshot ws.Cells(CLng(rowKey), fileIdCol), rowsToRestore.Item(CStr(rowKey))
    Next rowKey
    CaptureFileIdSelectionSnapshot Application.Selection, ws
    MsgBox "U列「ファイルID」は管理システムの正式識別値です。" & vbCrLf & _
        "直接編集・貼り付けはできないため、変更前の値に戻しました。" & vbCrLf & _
        "CSV取込または同期マクロで更新してください。", _
        vbExclamation, "ファイルIDは直接編集できません"
    新ファイル基準表_ファイルID直接編集を復元 = _
        Not NewFileStandardTargetHasNonFileIdCells(Target, ws, fileIdCol)
    Exit Function

RestoreFailed:
    On Error Resume Next
    For Each rowKey In attemptedSnapshots.Keys
        RestoreFileIdCellFromSnapshot ws.Cells(CLng(rowKey), fileIdCol), attemptedSnapshots.Item(CStr(rowKey))
    Next rowKey
    On Error GoTo 0
    MsgBox "ファイルIDを安全に復元できませんでした。" & vbCrLf & _
        "データの一部だけを復元した状態で続行しないため、ブックを保存せず終了してください。", _
        vbCritical, "ファイルID編集制限の安全停止"
    新ファイル基準表_ファイルID直接編集を復元 = True

End Function

Private Function NewFileStandardCaptureFileIdSelection( _
    ByVal Target As Range, ByVal ws As Worksheet) As Object

    Dim snapshots As Object
    Dim fileIdCol As Long
    Dim selectedFileIdRange As Range, selectedCell As Range
    Dim rowKey As String

    Set snapshots = CreateObject("Scripting.Dictionary")
    snapshots.CompareMode = vbTextCompare
    If Target Is Nothing Or ws Is Nothing Then GoTo SafeExit
    If Not (Target.Worksheet Is ws) Then GoTo SafeExit
    fileIdCol = NewFileStandardFindUniqueHeaderColumn(ws, "ファイルID")
    If fileIdCol <= 0 Then GoTo SafeExit
    Set selectedFileIdRange = Application.Intersect(Target, ws.Columns(fileIdCol))
    If selectedFileIdRange Is Nothing Then GoTo SafeExit
    If selectedFileIdRange.Cells.CountLarge > 5000 Then GoTo SafeExit

    For Each selectedCell In selectedFileIdRange.Cells
        If selectedCell.Row >= 2 Then
            rowKey = CStr(selectedCell.Row)
            If Not snapshots.Exists(rowKey) Then snapshots.Add rowKey, CaptureFileIdCellSnapshot(selectedCell)
        End If
    Next selectedCell

SafeExit:
    Set NewFileStandardCaptureFileIdSelection = snapshots

End Function

Private Function CaptureFileIdCellSnapshot(ByVal targetCell As Range) As Object

    Dim snapshot As Object

    Set snapshot = CreateObject("Scripting.Dictionary")
    snapshot.Add "Formula", CStr(targetCell.Formula)
    snapshot.Add "NumberFormat", CStr(targetCell.NumberFormat)
    Set CaptureFileIdCellSnapshot = snapshot

End Function

Private Function FileIdCellMatchesSnapshot( _
    ByVal targetCell As Range, ByVal snapshot As Object) As Boolean

    If targetCell Is Nothing Or snapshot Is Nothing Then Exit Function
    FileIdCellMatchesSnapshot = _
        (StrComp(CStr(targetCell.Formula), CStr(snapshot.Item("Formula")), vbBinaryCompare) = 0)

End Function

Private Sub RestoreFileIdCellFromSnapshot( _
    ByVal targetCell As Range, ByVal snapshot As Object)

    If targetCell Is Nothing Or snapshot Is Nothing Then _
        Err.Raise vbObjectError + 7311, "RestoreFileIdCellFromSnapshot", "復元元のファイルID情報がありません。"
    targetCell.NumberFormat = CStr(snapshot.Item("NumberFormat"))
    targetCell.Formula = CStr(snapshot.Item("Formula"))
    If Not FileIdCellMatchesSnapshot(targetCell, snapshot) Or _
       StrComp(CStr(targetCell.NumberFormat), CStr(snapshot.Item("NumberFormat")), vbBinaryCompare) <> 0 Then _
        Err.Raise vbObjectError + 7312, "RestoreFileIdCellFromSnapshot", "ファイルIDの復元結果を確認できません。"

End Sub

Private Function UndoUnsafeFileIdEdit(ByVal Target As Range, ByVal ws As Worksheet) As Boolean

    On Error GoTo UndoFailed
    Application.Undo
    CaptureFileIdSelectionSnapshot Application.Selection, ws
    MsgBox "ファイルIDの変更前の値を安全に特定できないため、編集操作全体を取り消しました。" & vbCrLf & _
        "複数セルを貼り付けていた場合も、今回の貼り付けは適用していません。", _
        vbExclamation, "ファイルIDは直接編集できません"
    UndoUnsafeFileIdEdit = True
    Exit Function

UndoFailed:
    MsgBox "ファイルIDを安全に復元できませんでした。" & vbCrLf & _
        "データを推測で変更しないため、ブックを保存せず終了してください。", _
        vbCritical, "ファイルID編集制限の安全停止"
    UndoUnsafeFileIdEdit = True

End Function

Private Function NewFileStandardTargetHasNonFileIdCells( _
    ByVal Target As Range, ByVal ws As Worksheet, ByVal fileIdCol As Long) As Boolean

    Dim changedFileIdRange As Range

    If Target Is Nothing Or ws Is Nothing Or fileIdCol <= 0 Then Exit Function
    Set changedFileIdRange = Application.Intersect(Target, ws.Columns(fileIdCol))
    If changedFileIdRange Is Nothing Then
        NewFileStandardTargetHasNonFileIdCells = (Target.Cells.CountLarge > 0)
    Else
        NewFileStandardTargetHasNonFileIdCells = _
            (Target.Cells.CountLarge > changedFileIdRange.Cells.CountLarge)
    End If

End Function

Private Function NewFileStandardFindUniqueHeaderColumn( _
    ByVal ws As Worksheet, ByVal headerName As String) As Long

    Dim lastCol As Long, colNo As Long, matchCount As Long

    If ws Is Nothing Then Exit Function
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For colNo = 1 To lastCol
        If StrComp(Trim$(CStr(ws.Cells(1, colNo).Value2)), headerName, vbTextCompare) = 0 Then
            matchCount = matchCount + 1
            NewFileStandardFindUniqueHeaderColumn = colNo
        End If
    Next colNo
    If matchCount <> 1 Then NewFileStandardFindUniqueHeaderColumn = 0

End Function

Private Function ShouldReindexSerial(ByVal Target As Range, ByVal ws As Worksheet) As Boolean

    Dim monitorHeaders As Variant
    Dim monitorRange As Range
    Dim workRange As Range
    Dim colNo As Long
    Dim i As Long

    ShouldReindexSerial = False
    If Target Is Nothing Then Exit Function
    If ws Is Nothing Then Exit Function

    monitorHeaders = Array("タイトル", "分類名２", "分類名３", "年度（和暦）", "保存期間")

    For i = LBound(monitorHeaders) To UBound(monitorHeaders)
        colNo = GetHeaderColumnByCandidates(ws, GetReindexHeaderCandidates(CStr(monitorHeaders(i))))
        If colNo > 0 Then
            If monitorRange Is Nothing Then
                Set monitorRange = ws.Range(ws.Cells(2, colNo), ws.Cells(ws.Rows.Count, colNo))
            Else
                Set monitorRange = Union(monitorRange, ws.Range(ws.Cells(2, colNo), ws.Cells(ws.Rows.Count, colNo)))
            End If
        End If
    Next i

    If monitorRange Is Nothing Then Exit Function

    Set workRange = Intersect(Target, monitorRange)
    ShouldReindexSerial = Not (workRange Is Nothing)

End Function

Private Function GetReindexHeaderCandidates(ByVal headerName As String) As Variant

    Select Case headerName
        Case "分類名２", "第1ガイド", "第１ガイド"
            GetReindexHeaderCandidates = Array("第1ガイド", "第１ガイド", "分類名２", "分類名2")
        Case "分類名３", "第2ガイド", "第２ガイド"
            GetReindexHeaderCandidates = Array("第2ガイド", "第２ガイド", "分類名３", "分類名3")
        Case "タイトル", "個別フォルダー", "個別フォルダ"
            GetReindexHeaderCandidates = Array("個別フォルダー", "個別フォルダ", "タイトル")
        Case "年度（和暦）"
            GetReindexHeaderCandidates = Array("年度（和暦）", "年度(和暦)")
        Case Else
            GetReindexHeaderCandidates = Array(headerName)
    End Select

End Function

Private Function NormalizeReindexHeaderName(ByVal headerName As String) As String

    Dim normalized As String

    normalized = Trim$(CStr(headerName))
    normalized = Replace(normalized, " ", "")
    normalized = Replace(normalized, "　", "")

    Select Case normalized
        Case "分類名2", "分類名２", "第1ガイド", "第１ガイド"
            NormalizeReindexHeaderName = "分類名２"
        Case "分類名3", "分類名３", "第2ガイド", "第２ガイド"
            NormalizeReindexHeaderName = "分類名３"
        Case "タイトル", "個別フォルダー", "個別フォルダ"
            NormalizeReindexHeaderName = "タイトル"
        Case Else
            NormalizeReindexHeaderName = normalized
    End Select

End Function

Private Function GetHeaderColumnByCandidates(ByVal ws As Worksheet, ByVal candidates As Variant) As Long

    Dim lastCol As Long
    Dim c As Long
    Dim i As Long
    Dim headerValue As String

    GetHeaderColumnByCandidates = 0
    If ws Is Nothing Then Exit Function

    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    If lastCol <= 0 Then Exit Function

    For c = 1 To lastCol
        headerValue = Trim(CStr(ws.Cells(1, c).Value))
        For i = LBound(candidates) To UBound(candidates)
            If StrComp(NormalizeReindexHeaderName(headerValue), NormalizeReindexHeaderName(CStr(candidates(i))), vbTextCompare) = 0 Then
                GetHeaderColumnByCandidates = c
                Exit Function
            End If
        Next i
    Next c

End Function
